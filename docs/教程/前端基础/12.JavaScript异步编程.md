# 第十二章 JavaScript异步编程

## 12.1 异步编程概述

### 12.1.1 什么是异步编程

JavaScript是单线程语言，意味着它一次只能执行一个任务。异步编程允许程序在等待某些操作（如网络请求、文件读取、定时器等）完成时，继续执行其他代码，而不是阻塞等待。

### 12.1.2 为什么需要异步编程

- **网络请求**：HTTP请求需要时间响应，不应阻塞UI
- **定时操作**：延迟执行或周期性执行代码
- **文件操作**：读写文件是I/O操作
- **用户交互**：响应用户操作不应被其他任务阻塞

### 12.1.3 异步编程的发展

JavaScript异步编程经历了几个发展阶段：

1. **回调函数**：最早的异步处理方式
2. **Promise**：ES6引入，更优雅的异步处理
3. **async/await**：ES2017引入，基于Promise的语法糖

## 12.2 回调函数

### 12.2.1 回调函数基础

回调函数是作为参数传递给另一个函数的函数，在异步操作完成后执行。

```javascript
// 模拟异步操作
function fetchData(callback) {
    setTimeout(function() {
        let data = { id: 1, name: 'John' };
        callback(data);
    }, 1000);
}

// 使用回调函数
fetchData(function(data) {
    console.log('获取到数据：', data);
});

console.log('这行代码会先执行');
```

### 12.2.2 回调地狱

当多个异步操作需要顺序执行时，回调函数会嵌套多层，形成"回调地狱"：

```javascript
getUser(function(user) {
    getOrders(user.id, function(orders) {
        getOrderDetails(orders[0].id, function(details) {
            getProduct(details.productId, function(product) {
                console.log(product);
            });
        });
    });
});
```

### 12.2.3 错误处理

回调函数的错误处理通常采用错误优先的回调风格：

```javascript
function fetchData(callback) {
    setTimeout(function() {
        let error = null;
        let data = null;
        
        // 模拟可能的错误
        if (Math.random() > 0.5) {
            error = new Error('获取数据失败');
        } else {
            data = { id: 1, name: 'John' };
        }
        
        callback(error, data);
    }, 1000);
}

fetchData(function(error, data) {
    if (error) {
        console.error('错误：', error.message);
        return;
    }
    console.log('数据：', data);
});
```

## 12.3 Promise

### 12.3.1 Promise概述

Promise是ES6引入的异步编程解决方案，代表一个尚未完成但预期将来会完成的操作。Promise有三种状态：

- **Pending（待定）**：初始状态，既未完成也未拒绝
- **Fulfilled（已完成）**：操作成功完成
- **Rejected（已拒绝）**：操作失败

### 12.3.2 创建Promise

```javascript
let promise = new Promise(function(resolve, reject) {
    // 异步操作
    setTimeout(function() {
        let success = true;
        
        if (success) {
            resolve('操作成功');
        } else {
            reject(new Error('操作失败'));
        }
    }, 1000);
});
```

### 12.3.3 使用Promise

```javascript
promise
    .then(function(result) {
        console.log('成功：', result);
    })
    .catch(function(error) {
        console.error('失败：', error.message);
    })
    .finally(function() {
        console.log('无论成功失败都执行');
    });
```

### 12.3.4 Promise链式调用

```javascript
function getUser(id) {
    return new Promise((resolve) => {
        setTimeout(() => resolve({ id, name: 'John' }), 1000);
    });
}

function getOrders(userId) {
    return new Promise((resolve) => {
        setTimeout(() => resolve([{ id: 1, total: 100 }]), 1000);
    });
}

// 链式调用
getUser(1)
    .then(user => {
        console.log('用户：', user);
        return getOrders(user.id);
    })
    .then(orders => {
        console.log('订单：', orders);
    })
    .catch(error => {
        console.error('错误：', error);
    });
```

### 12.3.5 Promise静态方法

**Promise.all**

等待所有Promise完成：

```javascript
let promise1 = fetch('/api/users');
let promise2 = fetch('/api/products');
let promise3 = fetch('/api/orders');

Promise.all([promise1, promise2, promise3])
    .then(responses => {
        console.log('所有请求完成');
        return Promise.all(responses.map(r => r.json()));
    })
    .then(data => {
        console.log('数据：', data);
    })
    .catch(error => {
        console.error('任一请求失败：', error);
    });
```

**Promise.race**

返回最先完成的Promise：

```javascript
let promise1 = new Promise(resolve => setTimeout(resolve, 500, '第一个'));
let promise2 = new Promise(resolve => setTimeout(resolve, 1000, '第二个'));

Promise.race([promise1, promise2])
    .then(result => {
        console.log('最快完成：', result);  // "第一个"
    });
```

**Promise.allSettled**

等待所有Promise完成（无论成功或失败）：

```javascript
let promises = [
    Promise.resolve('成功1'),
    Promise.reject('失败'),
    Promise.resolve('成功2')
];

Promise.allSettled(promises)
    .then(results => {
        results.forEach(result => {
            if (result.status === 'fulfilled') {
                console.log('成功：', result.value);
            } else {
                console.log('失败：', result.reason);
            }
        });
    });
```

**Promise.any**

返回第一个成功的Promise：

```javascript
let promises = [
    Promise.reject('错误1'),
    Promise.resolve('成功'),
    Promise.reject('错误2')
];

Promise.any(promises)
    .then(result => {
        console.log('第一个成功：', result);  // "成功"
    })
    .catch(error => {
        console.log('全部失败');
    });
```

## 12.4 async/await

### 12.4.1 async函数

`async`关键字用于声明异步函数，函数会自动返回一个Promise：

```javascript
async function fetchData() {
    return '数据';
}

// 等价于
function fetchData() {
    return Promise.resolve('数据');
}

fetchData().then(data => console.log(data));
```

### 12.4.2 await关键字

`await`关键字只能在`async`函数内部使用，用于等待Promise完成：

```javascript
async function getUserData() {
    try {
        let user = await getUser(1);
        console.log('用户：', user);
        
        let orders = await getOrders(user.id);
        console.log('订单：', orders);
        
        return { user, orders };
    } catch (error) {
        console.error('错误：', error);
        throw error;
    }
}

getUserData();
```

### 12.4.3 并行执行

```javascript
async function fetchAllData() {
    // 顺序执行（慢）
    // let users = await fetch('/api/users');
    // let products = await fetch('/api/products');
    
    // 并行执行（快）
    let [users, products, orders] = await Promise.all([
        fetch('/api/users').then(r => r.json()),
        fetch('/api/products').then(r => r.json()),
        fetch('/api/orders').then(r => r.json())
    ]);
    
    return { users, products, orders };
}
```

### 12.4.4 循环中的await

```javascript
// 顺序执行
async function processSequentially(items) {
    for (let item of items) {
        await processItem(item);  // 等待每个完成
    }
}

// 并行执行
async function processInParallel(items) {
    let promises = items.map(item => processItem(item));
    await Promise.all(promises);
}
```

## 12.5 Fetch API

### 12.5.1 基本使用

Fetch API是现代浏览器内置的HTTP请求API，基于Promise：

```javascript
// GET请求
fetch('https://api.example.com/users')
    .then(response => {
        if (!response.ok) {
            throw new Error('HTTP错误：' + response.status);
        }
        return response.json();
    })
    .then(data => {
        console.log('数据：', data);
    })
    .catch(error => {
        console.error('错误：', error);
    });

// async/await方式
async function getUsers() {
    try {
        let response = await fetch('https://api.example.com/users');
        
        if (!response.ok) {
            throw new Error('HTTP错误：' + response.status);
        }
        
        let data = await response.json();
        return data;
    } catch (error) {
        console.error('错误：', error);
        throw error;
    }
}
```

### 12.5.2 POST请求

```javascript
async function createUser(userData) {
    let response = await fetch('https://api.example.com/users', {
        method: 'POST',
        headers: {
            'Content-Type': 'application/json'
        },
        body: JSON.stringify(userData)
    });
    
    if (!response.ok) {
        throw new Error('创建失败');
    }
    
    return await response.json();
}

createUser({ name: 'John', email: 'john@example.com' })
    .then(user => console.log('创建成功：', user));
```

### 12.5.3 请求选项

```javascript
fetch('https://api.example.com/data', {
    method: 'POST',           // HTTP方法
    headers: {                // 请求头
        'Content-Type': 'application/json',
        'Authorization': 'Bearer token123'
    },
    body: JSON.stringify({    // 请求体
        key: 'value'
    }),
    mode: 'cors',             // 请求模式
    cache: 'no-cache',        // 缓存策略
    credentials: 'include'    // 发送cookie
});
```

## 12.6 定时器

### 12.6.1 setTimeout

延迟执行代码：

```javascript
// 基本用法
let timeoutId = setTimeout(function() {
    console.log('1秒后执行');
}, 1000);

// 带参数
setTimeout((name) => {
    console.log('Hello, ' + name);
}, 1000, 'John');

// 取消定时器
clearTimeout(timeoutId);

// Promise封装
function delay(ms) {
    return new Promise(resolve => setTimeout(resolve, ms));
}

async function example() {
    console.log('开始');
    await delay(1000);
    console.log('1秒后');
}
```

### 12.6.2 setInterval

周期性执行代码：

```javascript
// 基本用法
let intervalId = setInterval(function() {
    console.log('每秒执行一次');
}, 1000);

// 取消定时器
clearInterval(intervalId);

// 限制执行次数
let count = 0;
let interval = setInterval(() => {
    count++;
    console.log('执行第' + count + '次');
    
    if (count >= 5) {
        clearInterval(interval);
    }
}, 1000);
```

### 12.6.3 requestAnimationFrame

用于动画的定时器，与屏幕刷新率同步：

```javascript
function animate() {
    // 执行动画
    updatePosition();
    render();
    
    // 继续下一帧
    requestAnimationFrame(animate);
}

requestAnimationFrame(animate);

// 取消动画
let animationId = requestAnimationFrame(animate);
cancelAnimationFrame(animationId);
```
