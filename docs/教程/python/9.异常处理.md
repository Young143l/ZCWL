# 第九章 异常处理

## 9.1 异常概述

异常是程序运行过程中发生的不正常事件，会中断程序的正常执行流程。Python提供了完善的异常处理机制，帮助开发者捕获和处理异常，提高程序的健壮性。

Python中的异常都是BaseException类的子类，常见的异常类型包括：

| 异常类型 | 说明 |
|----------|------|
| SyntaxError | 语法错误 |
| NameError | 未声明的变量 |
| TypeError | 类型错误 |
| ValueError | 值错误 |
| IndexError | 索引越界 |
| KeyError | 字典键不存在 |
| AttributeError | 对象没有该属性 |
| ZeroDivisionError | 除零错误 |
| FileNotFoundError | 文件不存在 |
| IOError | 输入输出错误 |

## 9.2 异常处理结构

### try-except语句

```python
try:
    # 可能引发异常的代码
    result = 10 / 0
except ZeroDivisionError:
    # 处理除零异常
    print("除数不能为零")
```

### 捕获多种异常

```python
try:
    # 可能引发多种异常的代码
    num = int(input("请输入一个数字："))
    result = 100 / num
except ValueError:
    print("输入的不是有效数字")
except ZeroDivisionError:
    print("不能除以零")
```

### 捕获多个异常

```python
try:
    # 代码
    pass
except (ValueError, TypeError) as e:
    print(f"发生错误：{e}")
```

### 获取异常信息

```python
try:
    x = 1 / 0
except ZeroDivisionError as e:
    print(f"错误类型：{type(e).__name__}")
    print(f"错误信息：{e}")
```

### else子句

当没有异常发生时执行：

```python
try:
    num = int(input("请输入一个数字："))
    result = 100 / num
except ValueError:
    print("输入无效")
except ZeroDivisionError:
    print("不能除以零")
else:
    print(f"结果是：{result}")
```

### finally子句

无论是否发生异常都会执行：

```python
try:
    file = open("data.txt", "r")
    content = file.read()
except FileNotFoundError:
    print("文件不存在")
finally:
    print("清理资源")
    file.close()        # 确保文件被关闭
```

### 完整结构

```python
try:
    # 可能引发异常的代码
    pass
except ValueError as e:
    # 处理ValueError
    pass
except TypeError as e:
    # 处理TypeError
    pass
except Exception as e:
    # 处理其他所有异常
    pass
else:
    # 没有异常时执行
    pass
finally:
    # 无论有无异常都执行
    pass
```

## 9.3 常见异常处理

### 除零错误

```python
def safe_divide(a, b):
    try:
        return a / b
    except ZeroDivisionError:
        print("警告：除数为零，返回None")
        return None

result = safe_divide(10, 0)
print(result)           # None
```

### 类型错误

```python
def add_numbers(a, b):
    try:
        return a + b
    except TypeError:
        # 尝试转换类型
        try:
            return int(a) + int(b)
        except ValueError:
            print("无法转换为数字")
            return None

print(add_numbers("5", "3"))        # 8
print(add_numbers("abc", "3"))      # None
```

### 索引越界

```python
def safe_get(lst, index, default=None):
    try:
        return lst[index]
    except IndexError:
        return default

numbers = [1, 2, 3]
print(safe_get(numbers, 1))         # 2
print(safe_get(numbers, 10))        # None
print(safe_get(numbers, 10, -1))    # -1
```

### 键不存在

```python
def safe_get_dict(d, key, default=None):
    try:
        return d[key]
    except KeyError:
        return default

# 或使用get方法
person = {"name": "Alice", "age": 20}
print(person.get("name"))           # Alice
print(person.get("gender"))         # None
print(person.get("gender", "Unknown"))  # Unknown
```

## 9.4 抛出异常

### raise语句

```python
def validate_age(age):
    if not isinstance(age, int):
        raise TypeError("年龄必须是整数")
    if age < 0:
        raise ValueError("年龄不能为负数")
    if age > 150:
        raise ValueError("年龄不能超过150")
    return True

try:
    validate_age(-5)
except ValueError as e:
    print(f"验证失败：{e}")
```

### 自定义异常

```python
class ValidationError(Exception):
    """验证错误基类"""
    pass

class InvalidAgeError(ValidationError):
    """无效年龄错误"""
    def __init__(self, age, message="无效的年龄"):
        self.age = age
        self.message = message
        super().__init__(self.message)
    
    def __str__(self):
        return f"{self.message}: {self.age}"

class InvalidNameError(ValidationError):
    """无效姓名错误"""
    pass

def validate_person(name, age):
    if not name or len(name) < 2:
        raise InvalidNameError("姓名至少需要2个字符")
    if age < 0 or age > 150:
        raise InvalidAgeError(age)

try:
    validate_person("A", 200)
except InvalidNameError as e:
    print(f"姓名错误：{e}")
except InvalidAgeError as e:
    print(f"年龄错误：{e}")
```

## 9.5 异常链

### 使用from保留异常上下文

```python
def read_config(filename):
    try:
        with open(filename, 'r') as f:
            return f.read()
    except FileNotFoundError as e:
        raise ConfigError(f"无法读取配置文件：{filename}") from e

class ConfigError(Exception):
    pass

try:
    config = read_config("config.ini")
except ConfigError as e:
    print(f"错误：{e}")
    print(f"原始错误：{e.__cause__}")
```

### 使用raise from None

```python
def parse_number(s):
    try:
        return int(s)
    except ValueError:
        raise ValueError(f"'{s}' 不是有效的数字") from None

try:
    parse_number("abc")
except ValueError as e:
    print(e)        # 'abc' 不是有效的数字
```

## 9.7 上下文管理器

### with语句

```python
# 文件操作
with open("file.txt", "r") as f:
    content = f.read()
# 文件自动关闭

# 多个上下文
with open("input.txt", "r") as infile, \
     open("output.txt", "w") as outfile:
    outfile.write(infile.read())
```

### 自定义上下文管理器

```python
from contextlib import contextmanager

@contextmanager
def managed_resource(name):
    print(f"获取资源：{name}")
    resource = f"Resource-{name}"
    try:
        yield resource
    finally:
        print(f"释放资源：{name}")

with managed_resource("database") as db:
    print(f"使用{db}")

# 输出：
# 获取资源：database
# 使用Resource-database
# 释放资源：database
```

### 类形式的上下文管理器

```python
class DatabaseConnection:
    def __init__(self, connection_string):
        self.connection_string = connection_string
        self.connection = None
    
    def __enter__(self):
        print(f"连接到数据库：{self.connection_string}")
        self.connection = f"Connection-{self.connection_string}"
        return self.connection
    
    def __exit__(self, exc_type, exc_val, exc_tb):
        print("关闭数据库连接")
        if exc_type:
            print(f"发生异常：{exc_type.__name__}: {exc_val}")
        return False        # 不抑制异常

with DatabaseConnection("localhost:5432") as conn:
    print(f"使用连接：{conn}")
```

## 9.8 异常处理最佳实践

### 1. 捕获具体异常

```python
# 不好的做法
try:
    # 代码
    pass
except:                     # 捕获所有异常，包括KeyboardInterrupt
    pass

# 好的做法
try:
    # 代码
    pass
except ValueError as e:     # 捕获具体异常
    pass
```

### 2. 不要忽略异常

```python
# 不好的做法
try:
    do_something()
except SomeException:
    pass                    # 完全忽略异常

# 好的做法
try:
    do_something()
except SomeException as e:
    logger.error(f"操作失败：{e}")
    # 或重新抛出
    raise
```

### 3. 使用finally清理资源

```python
# 不好的做法
file = open("data.txt", "r")
try:
    content = file.read()
except FileNotFoundError:
    print("文件不存在")
# 如果发生其他异常，文件不会关闭

# 好的做法
try:
    file = open("data.txt", "r")
    content = file.read()
except FileNotFoundError:
    print("文件不存在")
finally:
    file.close()

# 更好的做法：使用with语句
with open("data.txt", "r") as file:
    content = file.read()
```

### 4. 提供有用的错误信息

```python
# 不好的做法
raise ValueError

# 好的做法
raise ValueError(f"年龄必须在0到150之间，得到：{age}")
```

### 5. 使用自定义异常

```python
class BusinessError(Exception):
    """业务逻辑错误"""
    pass

class ValidationError(BusinessError):
    """验证错误"""
    pass

class NotFoundError(BusinessError):
    """资源不存在错误"""
```

## 9.9 综合示例

### 示例1：安全计算器

```python
class CalculatorError(Exception):
    """计算器错误"""
    pass

class DivisionByZeroError(CalculatorError):
    """除零错误"""
    pass

class InvalidOperationError(CalculatorError):
    """无效操作错误"""
    pass

class SafeCalculator:
    """安全计算器"""
    
    def add(self, a, b):
        try:
            return a + b
        except TypeError:
            raise InvalidOperationError(f"无法相加：{a} 和 {b}")
    
    def subtract(self, a, b):
        try:
            return a - b
        except TypeError:
            raise InvalidOperationError(f"无法相减：{a} 和 {b}")
    
    def multiply(self, a, b):
        try:
            return a * b
        except TypeError:
            raise InvalidOperationError(f"无法相乘：{a} 和 {b}")
    
    def divide(self, a, b):
        try:
            if b == 0:
                raise DivisionByZeroError("除数不能为零")
            return a / b
        except TypeError:
            raise InvalidOperationError(f"无法相除：{a} 和 {b}")
    
    def calculate(self, expression):
        """计算表达式"""
        try:
            parts = expression.split()
            if len(parts) != 3:
                raise InvalidOperationError("表达式格式错误")
            
            a = float(parts[0])
            operator = parts[1]
            b = float(parts[2])
            
            operations = {
                "+": self.add,
                "-": self.subtract,
                "*": self.multiply,
                "/": self.divide
            }
            
            if operator not in operations:
                raise InvalidOperationError(f"不支持的运算符：{operator}")
            
            return operations[operator](a, b)
        
        except ValueError:
            raise InvalidOperationError("无效的数字格式")

# 使用
calc = SafeCalculator()

test_cases = [
    "10 + 5",
    "10 / 0",
    "10 / 2",
    "abc + 5",
    "10 ^ 2"
]

for expr in test_cases:
    try:
        result = calc.calculate(expr)
        print(f"{expr} = {result}")
    except CalculatorError as e:
        print(f"{expr} -> 错误：{e}")
```

### 示例2：文件处理器

```python
import os

class FileProcessorError(Exception):
    """文件处理错误"""
    pass

class FileProcessor:
    """文件处理器"""
    
    def read_file(self, filename):
        """安全读取文件"""
        try:
            with open(filename, 'r', encoding='utf-8') as f:
                return f.read()
        except FileNotFoundError:
            raise FileProcessorError(f"文件不存在：{filename}")
        except PermissionError:
            raise FileProcessorError(f"没有权限读取文件：{filename}")
        except UnicodeDecodeError as e:
            raise FileProcessorError(f"文件编码错误：{e}")
    
    def write_file(self, filename, content):
        """安全写入文件"""
        try:
            # 确保目录存在
            directory = os.path.dirname(filename)
            if directory and not os.path.exists(directory):
                os.makedirs(directory)
            
            with open(filename, 'w', encoding='utf-8') as f:
                f.write(content)
        
        except PermissionError:
            raise FileProcessorError(f"没有权限写入文件：{filename}")
        except IOError as e:
            raise FileProcessorError(f"写入文件失败：{e}")
    
    def copy_file(self, source, destination):
        """安全复制文件"""
        try:
            content = self.read_file(source)
            self.write_file(destination, content)
        except FileProcessorError:
            raise
        except Exception as e:
            raise FileProcessorError(f"复制文件失败：{e}")

# 使用
processor = FileProcessor()

try:
    content = processor.read_file("test.txt")
    print(f"文件内容：{content}")
except FileProcessorError as e:
    print(f"读取失败：{e}")

try:
    processor.write_file("output/test.txt", "Hello, World!")
    print("写入成功")
except FileProcessorError as e:
    print(f"写入失败：{e}")
```
