## 12.1 标准库概述

Python拥有丰富的标准库，涵盖了文件处理、网络编程、数据库接口、图形系统、数学运算等众多领域。标准库随Python一起安装，无需额外安装即可使用。

本章介绍一些最常用的标准库模块。

## 12.3 collections模块

### Counter

```python
from collections import Counter

# 计数
words = ["apple", "banana", "apple", "cherry", "banana", "apple"]
word_counts = Counter(words)
print(word_counts)                  # Counter({'apple': 3, 'banana': 2, 'cherry': 1})

# 获取最常见的元素
print(word_counts.most_common(2))   # [('apple', 3), ('banana', 2)]

# 计数操作
print(word_counts["apple"])         # 3
print(word_counts["orange"])        # 0

# 更新计数
word_counts.update(["apple", "orange"])
print(word_counts)

# 数学运算
c1 = Counter(a=3, b=1)
c2 = Counter(a=1, b=2)
print(c1 + c2)                      # Counter({'a': 4, 'b': 3})
print(c1 - c2)                      # Counter({'a': 2})
```

### defaultdict

```python
from collections import defaultdict

# 默认值为列表
d = defaultdict(list)
d["fruits"].append("apple")
d["fruits"].append("banana")
d["vegetables"].append("carrot")
print(dict(d))                      # {'fruits': ['apple', 'banana'], 'vegetables': ['carrot']}

# 默认值为整数
counts = defaultdict(int)
for word in ["apple", "banana", "apple"]:
    counts[word] += 1
print(dict(counts))                 # {'apple': 2, 'banana': 1}

# 自定义默认值
def default_value():
    return "unknown"

d = defaultdict(default_value)
print(d["key"])                     # unknown
```

### OrderedDict

```python
from collections import OrderedDict

# 有序字典（Python 3.7+普通字典也保持插入顺序）
od = OrderedDict()
od["a"] = 1
od["b"] = 2
od["c"] = 3

# 移动到末尾
od.move_to_end("a")
print(list(od.keys()))              # ['b', 'c', 'a']

# 移动到开头
od.move_to_end("a", last=False)
print(list(od.keys()))              # ['a', 'b', 'c']

# 弹出
last = od.popitem()                 # ('c', 3)
first = od.popitem(last=False)      # ('a', 1)
```

### namedtuple

```python
from collections import namedtuple

# 创建命名元组类
Point = namedtuple("Point", ["x", "y"])
p = Point(3, 4)

# 访问字段
print(p.x)                          # 3
print(p.y)                          # 4
print(p[0])                         # 3
print(p[1])                         # 4

# 不可变性
# p.x = 5                           # 错误：AttributeError

# 转换为字典
print(p._asdict())                  # {'x': 3, 'y': 4}

# 替换字段
p2 = p._replace(x=5)
print(p2)                           # Point(x=5, y=4)

# 默认值（Python 3.7+）
Person = namedtuple("Person", ["name", "age", "gender"], defaults=["unknown", 0, "unknown"])
p = Person("Alice")
print(p)                            # Person(name='Alice', age=0, gender='unknown')
```

### deque

```python
from collections import deque

# 双端队列
d = deque([1, 2, 3])

# 添加元素
d.append(4)                         # 右侧添加
d.appendleft(0)                     # 左侧添加
print(d)                            # deque([0, 1, 2, 3, 4])

# 删除元素
d.pop()                             # 右侧删除
d.popleft()                         # 左侧删除
print(d)                            # deque([1, 2, 3])

# 旋转
d = deque([1, 2, 3, 4, 5])
d.rotate(2)                         # 向右旋转2位
print(d)                            # deque([4, 5, 1, 2, 3])
d.rotate(-2)                        # 向左旋转2位
print(d)                            # deque([1, 2, 3, 4, 5])

# 限制大小
d = deque(maxlen=3)
d.append(1)
d.append(2)
d.append(3)
print(d)                            # deque([1, 2, 3])
d.append(4)                         # 超出大小，左侧元素被移除
print(d)                            # deque([2, 3, 4])
```

## 12.5 json模块

```python
import json

data = {
    "name": "Alice",
    "age": 20,
    "hobbies": ["reading", "swimming"],
    "address": {
        "city": "Beijing",
        "zipcode": "100000"
    }
}

# Python对象转JSON字符串
json_str = json.dumps(data)
print(json_str)

# 格式化输出
json_str = json.dumps(data, indent=4, ensure_ascii=False)
print(json_str)

# JSON字符串转Python对象
parsed = json.loads(json_str)
print(parsed)

# 读写JSON文件
with open("data.json", "w", encoding="utf-8") as f:
    json.dump(data, f, indent=4, ensure_ascii=False)

with open("data.json", "r", encoding="utf-8") as f:
    loaded = json.load(f)
    print(loaded)

# 自定义编码器
class Person:
    def __init__(self, name, age):
        self.name = name
        self.age = age

class PersonEncoder(json.JSONEncoder):
    def default(self, obj):
        if isinstance(obj, Person):
            return {"name": obj.name, "age": obj.age}
        return super().default(obj)

person = Person("Alice", 20)
json_str = json.dumps(person, cls=PersonEncoder)
print(json_str)
```

## 12.6 re模块

```python
import re

# 匹配
pattern = r"\d+"
text = "There are 123 apples and 456 oranges"
match = re.search(pattern, text)
if match:
    print(match.group())            # 123
    print(match.start())            # 10
    print(match.end())              # 13

# 查找所有
matches = re.findall(pattern, text)
print(matches)                      # ['123', '456']

# 替换
result = re.sub(pattern, "X", text)
print(result)                       # There are X apples and X oranges

# 分割
parts = re.split(r"\s+", "Hello   World  Python")
print(parts)                        # ['Hello', 'World', 'Python']

# 编译正则表达式
pattern = re.compile(r"(\d{4})-(\d{2})-(\d{2})")
match = pattern.search("Date: 2024-01-15")
if match:
    print(match.group(0))           # 2024-01-15
    print(match.group(1))           # 2024
    print(match.group(2))           # 01
    print(match.group(3))           # 15

# 常用模式
email_pattern = r"^[\w\.-]+@[\w\.-]+\.\w+$"
print(re.match(email_pattern, "test@example.com"))      # Match对象
print(re.match(email_pattern, "invalid"))               # None
```

## 12.7 random模块

```python
import random

# 随机整数
print(random.randint(1, 10))        # 1到10的随机整数
print(random.randrange(1, 10, 2))   # 1到9的奇数

# 随机浮点数
print(random.random())              # 0到1的随机浮点数
print(random.uniform(1, 10))        # 1到10的随机浮点数

# 随机选择
choices = ["apple", "banana", "cherry"]
print(random.choice(choices))
print(random.choices(choices, k=2))     # 可重复选择
print(random.sample(choices, k=2))      # 不重复选择

# 打乱顺序
lst = [1, 2, 3, 4, 5]
random.shuffle(lst)
print(lst)

# 设置随机种子
random.seed(42)
print(random.random())              # 可重现的随机数
```

## 12.9 functools模块

```python
from functools import lru_cache, partial, reduce, wraps

# lru_cache：缓存函数结果
@lru_cache(maxsize=None)
def fibonacci(n):
    if n < 2:
        return n
    return fibonacci(n - 1) + fibonacci(n - 2)

print(fibonacci(100))                   # 快速计算

# partial：偏函数
basetwo = partial(int, base=2)
print(basetwo("1010"))                  # 10

# reduce：累积计算
numbers = [1, 2, 3, 4, 5]
product = reduce(lambda x, y: x * y, numbers)
print(product)                          # 120

# wraps：保留原函数元信息
def my_decorator(func):
    @wraps(func)
    def wrapper(*args, **kwargs):
        print("Before")
        result = func(*args, **kwargs)
        print("After")
        return result
    return wrapper

@my_decorator
def greet():
    """问候函数"""
    print("Hello!")

print(greet.__name__)                   # greet
print(greet.__doc__)                    # 问候函数
```

## 12.10 itertools模块

```python
import itertools

# 无限迭代器
counter = itertools.count(10, 2)
for _ in range(5):
    print(next(counter))                # 10, 12, 14, 16, 18

# 链式迭代
for item in itertools.chain([1, 2, 3], [4, 5, 6]):
    print(item)

# 组合
for item in itertools.combinations([1, 2, 3], 2):
    print(item)                         # (1, 2), (1, 3), (2, 3)

# 排列
for item in itertools.permutations([1, 2, 3], 2):
    print(item)                         # (1, 2), (1, 3), (2, 1), (2, 3), (3, 1), (3, 2)

# 笛卡尔积
for item in itertools.product([1, 2], ["a", "b"]):
    print(item)                         # (1, 'a'), (1, 'b'), (2, 'a'), (2, 'b')

# 分组
data = [("A", 1), ("A", 2), ("B", 3)]
for key, group in itertools.groupby(data, lambda x: x[0]):
    print(key, list(group))
```

## 12.11 综合示例

### 示例1：数据处理器

```python
import json
import csv
from datetime import datetime
from collections import Counter, defaultdict

class DataProcessor:
    """数据处理器"""
    
    def __init__(self):
        self.data = []
    
    def load_json(self, filename):
        """加载JSON文件"""
        with open(filename, "r", encoding="utf-8") as f:
            self.data = json.load(f)
    
    def load_csv(self, filename):
        """加载CSV文件"""
        with open(filename, "r", encoding="utf-8") as f:
            reader = csv.DictReader(f)
            self.data = list(reader)
    
    def save_json(self, filename):
        """保存为JSON文件"""
        with open(filename, "w", encoding="utf-8") as f:
            json.dump(self.data, f, indent=4, ensure_ascii=False)
    
    def filter_by(self, key, value):
        """按条件过滤"""
        return [item for item in self.data if item.get(key) == value]
    
    def group_by(self, key):
        """按字段分组"""
        groups = defaultdict(list)
        for item in self.data:
            groups[item.get(key)].append(item)
        return dict(groups)
    
    def count_by(self, key):
        """按字段计数"""
        return Counter(item.get(key) for item in self.data)
    
    def sort_by(self, key, reverse=False):
        """按字段排序"""
        return sorted(self.data, key=lambda x: x.get(key), reverse=reverse)

# 使用
processor = DataProcessor()
processor.data = [
    {"name": "Alice", "age": 20, "city": "Beijing"},
    {"name": "Bob", "age": 25, "city": "Shanghai"},
    {"name": "Charlie", "age": 20, "city": "Beijing"}
]

print(processor.count_by("city"))
print(processor.group_by("age"))
print(processor.sort_by("age", reverse=True))
```
