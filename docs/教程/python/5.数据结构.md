# 第五章 数据结构

## 5.1 数据结构概述

Python提供了多种内置数据结构，用于组织和存储数据。这些数据结构各有特点，适用于不同的场景。

| 数据结构 | 特点 | 适用场景 |
|----------|------|----------|
| 列表（list） | 有序、可变、可重复 | 需要有序存储和修改数据的场景 |
| 元组（tuple） | 有序、不可变、可重复 | 数据不需要修改的场景 |
| 字典（dict） | 键值对、无序、键唯一 | 通过键快速查找值的场景 |
| 集合（set） | 无序、不重复 | 去重和集合运算的场景 |

## 5.2 列表（list）

列表是Python中最常用的数据结构，是有序、可变的元素集合。

### 创建列表

```python
# 空列表
empty_list = []
empty_list = list()

# 包含元素的列表
numbers = [1, 2, 3, 4, 5]
mixed = [1, "hello", 3.14, True, [1, 2, 3]]

# 从其他可迭代对象创建
list_from_string = list("hello")        # ['h', 'e', 'l', 'l', 'o']
list_from_range = list(range(5))        # [0, 1, 2, 3, 4]
list_from_tuple = list((1, 2, 3))       # [1, 2, 3]
```

### 访问元素

```python
fruits = ["apple", "banana", "cherry", "date", "elderberry"]

# 索引访问
print(fruits[0])        # apple
print(fruits[-1])       # elderberry（最后一个）
print(fruits[-2])       # date（倒数第二个）

# 切片访问
print(fruits[1:4])      # ['banana', 'cherry', 'date']
print(fruits[:3])       # ['apple', 'banana', 'cherry']
print(fruits[2:])       # ['cherry', 'date', 'elderberry']
print(fruits[::2])      # ['apple', 'cherry', 'elderberry']（步长为2）
print(fruits[::-1])     # 反转列表
```

### 修改列表

```python
# 修改元素
fruits[0] = "apricot"

# 添加元素
fruits.append("fig")                    # 在末尾添加
fruits.insert(1, "blueberry")           # 在指定位置插入
fruits.extend(["grape", "honeydew"])    # 扩展列表

# 删除元素
del fruits[0]                           # 删除指定索引的元素
fruits.remove("banana")                 # 删除第一个匹配的元素
fruits.pop()                            # 删除并返回最后一个元素
fruits.pop(0)                           # 删除并返回指定索引的元素
fruits.clear()                          # 清空列表

# 其他操作
fruits.sort()                           # 原地排序
fruits.sort(reverse=True)               # 降序排序
fruits.reverse()                        # 原地反转
new_list = sorted(fruits)               # 返回排序后的新列表
new_list = list(reversed(fruits))       # 返回反转后的新列表
```

### 列表方法

```python
numbers = [3, 1, 4, 1, 5, 9, 2, 6]

# 长度
print(len(numbers))

# 计数
print(numbers.count(1))                 # 2

# 查找索引
print(numbers.index(5))                 # 4
print(numbers.index(1, 2))              # 从索引2开始查找

# 复制
copy1 = numbers.copy()
copy2 = numbers[:]
copy3 = list(numbers)
```

### 列表推导式

```python
# 基本形式
squares = [x**2 for x in range(10)]

# 带条件
even_squares = [x**2 for x in range(10) if x % 2 == 0]

# 多重循环
matrix = [[i*j for j in range(1, 4)] for i in range(1, 4)]

# 扁平化
nested = [[1, 2, 3], [4, 5, 6], [7, 8, 9]]
flat = [x for sublist in nested for x in sublist]
```

## 5.3 元组（tuple）

元组是有序、不可变的元素集合。

### 创建元组

```python
# 空元组
empty_tuple = ()
empty_tuple = tuple()

# 包含元素的元组
numbers = (1, 2, 3, 4, 5)
single = (1,)                           # 单元素元组需要逗号
no_parens = 1, 2, 3                     # 括号可省略

# 从其他可迭代对象创建
tuple_from_list = tuple([1, 2, 3])
tuple_from_string = tuple("hello")
```

### 访问元素

```python
t = ("apple", "banana", "cherry")

print(t[0])             # apple
print(t[-1])            # cherry
print(t[1:3])           # ('banana', 'cherry')
```

### 元组的不可变性

```python
t = (1, 2, 3)
# t[0] = 10             # 错误：TypeError

# 但可以修改可变元素
t = ([1, 2], 3)
t[0].append(3)          # 可以修改列表元素
```

### 元组解包

```python
# 基本解包
a, b, c = (1, 2, 3)

# 扩展解包
first, *rest = (1, 2, 3, 4, 5)
*beginning, last = (1, 2, 3, 4, 5)
first, *middle, last = (1, 2, 3, 4, 5)

# 交换变量
a, b = b, a

# 函数返回多个值
def get_min_max(numbers):
    return min(numbers), max(numbers)

min_val, max_val = get_min_max([3, 1, 4, 1, 5])
```

## 5.4 字典（dict）

字典是键值对的集合，键必须唯一且不可变。

### 创建字典

```python
# 空字典
empty_dict = {}
empty_dict = dict()

# 包含元素的字典
person = {"name": "Alice", "age": 20, "city": "Beijing"}

# 使用dict()创建
person = dict(name="Alice", age=20, city="Beijing")
person = dict([("name", "Alice"), ("age", 20)])

# 字典推导式
squares = {x: x**2 for x in range(6)}
```

### 访问元素

```python
person = {"name": "Alice", "age": 20, "city": "Beijing"}

# 使用键访问
print(person["name"])                   # Alice
# print(person["gender"])               # KeyError

# 使用get方法（安全访问）
print(person.get("name"))               # Alice
print(person.get("gender"))             # None
print(person.get("gender", "Unknown"))  # Unknown

# 检查键是否存在
if "name" in person:
    print("Name exists")
```

### 修改字典

```python
# 添加或修改
person["age"] = 21
person["gender"] = "Female"

# 使用update方法
person.update({"email": "alice@example.com", "phone": "1234567890"})

# 删除元素
del person["phone"]
age = person.pop("age")                 # 删除并返回值
item = person.popitem()                 # 删除并返回最后一个键值对
person.clear()                          # 清空字典
```

### 遍历字典

```python
person = {"name": "Alice", "age": 20, "city": "Beijing"}

# 遍历键
for key in person:
    print(key)

for key in person.keys():
    print(key)

# 遍历值
for value in person.values():
    print(value)

# 遍历键值对
for key, value in person.items():
    print(f"{key}: {value}")
```

### 字典方法

```python
person = {"name": "Alice", "age": 20}

# 长度
print(len(person))

# 合并字典（Python 3.9+）
other = {"city": "Beijing"}
merged = person | other

# 设置默认值
person.setdefault("gender", "Unknown")

# 从键创建字典
keys = ["a", "b", "c"]
d = dict.fromkeys(keys, 0)              # {'a': 0, 'b': 0, 'c': 0}
```

## 5.5 集合（set）

集合是无序、不重复的元素集合。

### 创建集合

```python
# 空集合
empty_set = set()                       # 不能用{}，那是空字典

# 包含元素的集合
numbers = {1, 2, 3, 4, 5}

# 从其他可迭代对象创建
set_from_list = set([1, 2, 2, 3, 3, 3])     # {1, 2, 3}
set_from_string = set("hello")              # {'h', 'e', 'l', 'o'}

# 集合推导式
squares = {x**2 for x in range(10)}
```

### 修改集合

```python
s = {1, 2, 3}

# 添加元素
s.add(4)
s.update([5, 6, 7])                     # 添加多个元素

# 删除元素
s.remove(4)                             # 元素不存在会报错
s.discard(4)                            # 元素不存在不会报错
s.pop()                                 # 删除并返回任意元素
s.clear()                               # 清空集合
```

### 集合运算

```python
a = {1, 2, 3, 4}
b = {3, 4, 5, 6}

# 并集
print(a | b)                            # {1, 2, 3, 4, 5, 6}
print(a.union(b))

# 交集
print(a & b)                            # {3, 4}
print(a.intersection(b))

# 差集
print(a - b)                            # {1, 2}
print(a.difference(b))

# 对称差集
print(a ^ b)                            # {1, 2, 5, 6}
print(a.symmetric_difference(b))

# 子集和超集
print(a <= b)                           # 是否为子集
print(a.issubset(b))
print(a >= b)                           # 是否为超集
print(a.issuperset(b))

# 不相交
print(a.isdisjoint(b))                  # 是否没有共同元素
```

### 不可变集合（frozenset）

```python
fs = frozenset([1, 2, 3])
# fs.add(4)             # 错误：不可变
```

## 5.6 数据结构转换

```python
# 列表与元组互转
lst = [1, 2, 3]
t = tuple(lst)                          # (1, 2, 3)
lst2 = list(t)                          # [1, 2, 3]

# 列表与集合互转
lst = [1, 2, 2, 3, 3, 3]
s = set(lst)                            # {1, 2, 3}
lst2 = list(s)                          # [1, 2, 3]（顺序不保证）

# 列表与字典
lst = [("a", 1), ("b", 2)]
d = dict(lst)                           # {'a': 1, 'b': 2}
lst2 = list(d.items())                  # [('a', 1), ('b', 2)]

# 字符串与列表
s = "hello"
lst = list(s)                           # ['h', 'e', 'l', 'l', 'o']
s2 = "".join(lst)                       # "hello"
s3 = "-".join(lst)                      # "h-e-l-l-o"
```

## 5.7 综合示例

### 示例1：列表操作

```python
def list_operations():
    # 创建列表
    numbers = [3, 1, 4, 1, 5, 9, 2, 6, 5, 3, 5]
    
    # 基本统计
    print(f"列表：{numbers}")
    print(f"长度：{len(numbers)}")
    print(f"最大值：{max(numbers)}")
    print(f"最小值：{min(numbers)}")
    print(f"总和：{sum(numbers)}")
    print(f"平均值：{sum(numbers) / len(numbers):.2f}")
    
    # 去重并排序
    unique_sorted = sorted(set(numbers))
    print(f"去重排序后：{unique_sorted}")
    
    # 查找元素位置
    target = 5
    positions = [i for i, x in enumerate(numbers) if x == target]
    print(f"{target}出现在位置：{positions}")
    
    # 过滤
    evens = [x for x in numbers if x % 2 == 0]
    print(f"偶数：{evens}")

list_operations()
```

### 示例2：字典操作

```python
def word_frequency(text):
    """统计词频"""
    words = text.lower().split()
    frequency = {}
    
    for word in words:
        word = word.strip(".,!?;:'\"")
        frequency[word] = frequency.get(word, 0) + 1
    
    return frequency

def dict_operations():
    text = "the quick brown fox jumps over the lazy dog the fox was quick"
    freq = word_frequency(text)
    
    # 按频率排序
    sorted_freq = sorted(freq.items(), key=lambda x: x[1], reverse=True)
    
    print("词频统计：")
    for word, count in sorted_freq:
        print(f"  {word}: {count}")
    
    # 找出最高频的词
    most_common = max(freq.items(), key=lambda x: x[1])
    print(f"\n最高频的词：{most_common[0]}（出现{most_common[1]}次）")

dict_operations()
```

### 示例3：集合操作

```python
def set_operations():
    # 两个班级的学生
    class_a = {"Alice", "Bob", "Charlie", "David"}
    class_b = {"Bob", "David", "Eve", "Frank"}
    
    # 同时参加两个班级的学生
    both = class_a & class_b
    print(f"同时参加两个班级的学生：{both}")
    
    # 只参加一个班级的学生
    only_one = class_a ^ class_b
    print(f"只参加一个班级的学生：{only_one}")
    
    # 所有学生
    all_students = class_a | class_b
    print(f"所有学生：{all_students}")
    
    # 只参加A班不参加B班的学生
    only_a = class_a - class_b
    print(f"只参加A班的学生：{only_a}")
    
    # 检查是否为子集
    print(f"A班是否为所有学生的子集：{class_a <= all_students}")

set_operations()
```

### 示例4：学生成绩管理

```python
def student_management():
    # 学生成绩数据
    students = {
        "Alice": {"math": 85, "english": 90, "science": 88},
        "Bob": {"math": 78, "english": 82, "science": 80},
        "Charlie": {"math": 92, "english": 88, "science": 95},
        "David": {"math": 70, "english": 75, "science": 72}
    }
    
    # 计算每个学生的平均分
    print("学生成绩统计：")
    print("-" * 50)
    
    for name, scores in students.items():
        avg = sum(scores.values()) / len(scores)
        print(f"{name:10} 平均分：{avg:.2f}")
    
    # 找出每门课的最高分
    print("\n每门课最高分：")
    subjects = ["math", "english", "science"]
    
    for subject in subjects:
        max_score = max(students[name][subject] for name in students)
        top_students = [name for name in students 
                       if students[name][subject] == max_score]
        print(f"{subject:10} 最高分：{max_score} ({', '.join(top_students)})")
    
    # 按平均分排序
    avg_scores = [(name, sum(scores.values()) / len(scores)) 
                  for name, scores in students.items()]
    avg_scores.sort(key=lambda x: x[1], reverse=True)
    
    print("\n按平均分排名：")
    for rank, (name, avg) in enumerate(avg_scores, 1):
        print(f"{rank}. {name}: {avg:.2f}")

student_management()
```
