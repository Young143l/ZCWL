# 第十一章 迭代器与生成器

## 11.1 迭代器概述

迭代器是Python中用于遍历集合的对象。迭代器对象实现了迭代器协议，即包含`__iter__()`和`__next__()`方法。

迭代器的特点：
- 惰性求值：只在需要时生成值
- 只能遍历一次
- 节省内存

## 11.2 可迭代对象与迭代器

### 可迭代对象

可迭代对象是指实现了`__iter__()`方法或`__getitem__()`方法的对象：

```python
# 常见的可迭代对象
list_obj = [1, 2, 3]
tuple_obj = (1, 2, 3)
dict_obj = {"a": 1, "b": 2}
set_obj = {1, 2, 3}
str_obj = "hello"
range_obj = range(10)
file_obj = open("file.txt")

# 检查是否为可迭代对象
from collections.abc import Iterable

print(isinstance([1, 2, 3], Iterable))      # True
print(isinstance("hello", Iterable))        # True
print(isinstance(123, Iterable))            # False
```

### 迭代器对象

迭代器对象实现了`__iter__()`和`__next__()`方法：

```python
from collections.abc import Iterator

# 从可迭代对象获取迭代器
lst = [1, 2, 3]
iterator = iter(lst)

print(isinstance(iterator, Iterator))       # True

# 使用next()获取下一个元素
print(next(iterator))       # 1
print(next(iterator))       # 2
print(next(iterator))       # 3
# print(next(iterator))     # StopIteration异常
```

### 自定义迭代器

```python
class CountDown:
    """倒计时迭代器"""
    
    def __init__(self, start):
        self.start = start
    
    def __iter__(self):
        return self
    
    def __next__(self):
        if self.start <= 0:
            raise StopIteration
        self.start -= 1
        return self.start + 1

# 使用
countdown = CountDown(5)
for num in countdown:
    print(num)          # 5, 4, 3, 2, 1
```

## 11.3 生成器

生成器是创建迭代器的简洁方式，使用`yield`关键字。

### 生成器函数

```python
def countdown(n):
    """倒计时生成器"""
    while n > 0:
        yield n
        n -= 1

# 使用
for num in countdown(5):
    print(num)          # 5, 4, 3, 2, 1

# 获取生成器对象
gen = countdown(3)
print(next(gen))        # 3
print(next(gen))        # 2
print(next(gen))        # 1
# print(next(gen))      # StopIteration
```

### 生成器表达式

```python
# 列表推导式（创建完整列表）
squares_list = [x**2 for x in range(10)]

# 生成器表达式（惰性求值）
squares_gen = (x**2 for x in range(10))

print(type(squares_list))   # <class 'list'>
print(type(squares_gen))    # <class 'generator'>

# 使用生成器表达式
for square in squares_gen:
    print(square)

# 生成器表达式只能遍历一次
# for square in squares_gen:  # 不会输出任何内容
#     print(square)
```

### 生成器与列表比较

```python
import sys

# 列表
list_data = [x**2 for x in range(1000000)]
print(f"列表大小：{sys.getsizeof(list_data)} bytes")

# 生成器
gen_data = (x**2 for x in range(1000000))
print(f"生成器大小：{sys.getsizeof(gen_data)} bytes")

# 生成器节省内存，但只能遍历一次
```

## 11.4 生成器的高级用法

### 生成器状态

```python
def simple_generator():
    print("开始")
    yield 1
    print("继续")
    yield 2
    print("结束")

gen = simple_generator()
print(next(gen))        # 开始，然后输出 1
print(next(gen))        # 继续，然后输出 2
# print(next(gen))      # 结束，然后抛出 StopIteration
```

### send方法

```python
def accumulator():
    total = 0
    while True:
        value = yield total
        if value is None:
            break
        total += value

acc = accumulator()
next(acc)               # 启动生成器

print(acc.send(10))     # 10
print(acc.send(20))     # 30
print(acc.send(30))     # 60
acc.close()
```

### throw方法

```python
def generator_with_exception():
    try:
        yield "开始"
        yield "继续"
    except ValueError as e:
        yield f"捕获异常：{e}"
    yield "结束"

gen = generator_with_exception()
print(next(gen))                    # 开始
print(gen.throw(ValueError, "测试")) # 捕获异常：测试
print(next(gen))                    # 结束
```

### close方法

```python
def generator_with_cleanup():
    try:
        yield "开始"
        yield "继续"
    finally:
        print("清理资源")

gen = generator_with_cleanup()
print(next(gen))        # 开始
gen.close()             # 清理资源
```

### yield from

```python
# 嵌套生成器
def sub_generator():
    yield 1
    yield 2
    yield 3

def main_generator():
    yield "开始"
    yield from sub_generator()
    yield "结束"

for item in main_generator():
    print(item)
# 输出：开始, 1, 2, 3, 结束

# yield from处理子生成器的返回值
def average():
    total = 0
    count = 0
    while True:
        value = yield
        if value is None:
            break
        total += value
        count += 1
    return total / count

def delegate():
    result = yield from average()
    print(f"平均值：{result}")

del_gen = delegate()
next(del_gen)           # 启动
del_gen.send(10)
del_gen.send(20)
del_gen.send(30)
del_gen.send(None)      # 平均值：20.0
```

## 11.5 迭代工具

### itertools模块

```python
import itertools

# count：无限计数器
for i in itertools.count(10, 2):
    if i > 20:
        break
    print(i)            # 10, 12, 14, 16, 18, 20

# cycle：无限循环
for i, item in enumerate(itertools.cycle("ABC")):
    if i >= 6:
        break
    print(item)         # A, B, C, A, B, C

# repeat：重复
for item in itertools.repeat("Hello", 3):
    print(item)         # Hello, Hello, Hello

# chain：连接多个可迭代对象
for item in itertools.chain([1, 2, 3], [4, 5, 6]):
    print(item)         # 1, 2, 3, 4, 5, 6

# zip_longest：按最长序列组合
for item in itertools.zip_longest([1, 2, 3], ["a", "b"], fillvalue="-"):
    print(item)         # (1, 'a'), (2, 'b'), (3, '-')

# combinations：组合
for item in itertools.combinations([1, 2, 3], 2):
    print(item)         # (1, 2), (1, 3), (2, 3)

# permutations：排列
for item in itertools.permutations([1, 2, 3], 2):
    print(item)         # (1, 2), (1, 3), (2, 1), (2, 3), (3, 1), (3, 2)

# product：笛卡尔积
for item in itertools.product([1, 2], ["a", "b"]):
    print(item)         # (1, 'a'), (1, 'b'), (2, 'a'), (2, 'b')

# groupby：分组
data = [("A", 1), ("A", 2), ("B", 3), ("B", 4)]
for key, group in itertools.groupby(data, lambda x: x[0]):
    print(key, list(group))
# A [(A, 1), (A, 2)]
# B [(B, 3), (B, 4)]
```

### 内置迭代函数

```python
# enumerate：枚举
for index, value in enumerate(["a", "b", "c"], start=1):
    print(f"{index}: {value}")

# zip：并行迭代
for name, age in zip(["Alice", "Bob"], [20, 25]):
    print(f"{name}: {age}")

# map：映射
for result in map(lambda x: x**2, [1, 2, 3, 4]):
    print(result)

# filter：过滤
for result in filter(lambda x: x > 2, [1, 2, 3, 4]):
    print(result)

# reversed：反转
for item in reversed([1, 2, 3, 4]):
    print(item)

# sorted：排序
for item in sorted([3, 1, 4, 1, 5]):
    print(item)

# sum、min、max、all、any
print(sum([1, 2, 3, 4, 5]))
print(min([3, 1, 4, 1, 5]))
print(max([3, 1, 4, 1, 5]))
print(all([True, True, False]))
print(any([False, True, False]))
```

## 11.6 实际应用

### 示例1：文件逐行读取

```python
def read_large_file(file_path):
    """逐行读取大文件"""
    with open(file_path, "r", encoding="utf-8") as f:
        for line in f:
            yield line.strip()

# 使用
for line in read_large_file("large_file.txt"):
    # 处理每一行，不占用大量内存
    process(line)
```

### 示例2：无限序列

```python
def fibonacci():
    """无限斐波那契数列"""
    a, b = 0, 1
    while True:
        yield a
        a, b = b, a + b

# 使用
fib = fibonacci()
for _ in range(10):
    print(next(fib))
```

### 示例3：数据管道

```python
def data_source():
    """数据源"""
    for i in range(100):
        yield i

def filter_even(data):
    """过滤偶数"""
    for item in data:
        if item % 2 == 0:
            yield item

def multiply_by_ten(data):
    """乘以10"""
    for item in data:
        yield item * 10

# 构建数据管道
pipeline = multiply_by_ten(filter_even(data_source()))

for result in pipeline:
    print(result)       # 0, 20, 40, 60, ...
```

### 示例4：上下文管理器生成器

```python
from contextlib import contextmanager

@contextmanager
def managed_resource(name):
    print(f"获取资源：{name}")
    resource = f"Resource-{name}"
    try:
        yield resource
    finally:
        print(f"释放资源：{name}")

with managed_resource("database") as db:
    print(f"使用{db}")

# 输出：
# 获取资源：database
# 使用Resource-database
# 释放资源：database
```

### 示例5：递归生成器

```python
def tree_generator(node):
    """递归遍历树"""
    yield node.value
    for child in node.children:
        yield from tree_generator(child)

class TreeNode:
    def __init__(self, value):
        self.value = value
        self.children = []
    
    def add_child(self, child):
        self.children.append(child)

# 构建树
root = TreeNode("A")
root.add_child(TreeNode("B"))
root.add_child(TreeNode("C"))
root.children[0].add_child(TreeNode("D"))

# 遍历
for value in tree_generator(root):
    print(value)        # A, B, D, C
```
