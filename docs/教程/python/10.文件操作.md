# 第十章 文件操作

## 10.1 文件操作概述

文件是存储在持久化存储介质上的数据集合。Python提供了丰富的文件操作功能，包括文件的打开、读取、写入和关闭等操作。

文件按照内容可以分为：
- 文本文件：以字符形式存储，如.txt、.py、.csv等
- 二进制文件：以字节形式存储，如.jpg、.mp3、.exe等

## 10.2 打开和关闭文件

### open函数

```python
file = open(filename, mode, encoding=None)
```

模式参数：

| 模式 | 说明 |
|------|------|
| 'r' | 只读（默认） |
| 'w' | 只写，文件不存在则创建，存在则清空 |
| 'x' | 创建新文件，文件存在则报错 |
| 'a' | 追加，文件不存在则创建 |
| 'b' | 二进制模式 |
| 't' | 文本模式（默认） |
| '+' | 读写模式 |

组合模式：
- 'r+'：读写，文件必须存在
- 'w+'：读写，文件不存在则创建，存在则清空
- 'a+'：读写追加
- 'rb'、'wb'、'ab'：二进制模式

### 关闭文件

```python
file = open("test.txt", "r")
# 文件操作
file.close()
```

### 使用with语句

```python
with open("test.txt", "r") as file:
    # 文件操作
    pass
# 文件自动关闭
```

## 10.3 读取文件

### 读取整个文件

```python
with open("test.txt", "r", encoding="utf-8") as file:
    content = file.read()
    print(content)
```

### 按行读取

```python
# 方法1：readlines()
with open("test.txt", "r", encoding="utf-8") as file:
    lines = file.readlines()
    for line in lines:
        print(line.strip())

# 方法2：直接迭代
with open("test.txt", "r", encoding="utf-8") as file:
    for line in file:
        print(line.strip())

# 方法3：readline()
with open("test.txt", "r", encoding="utf-8") as file:
    while True:
        line = file.readline()
        if not line:
            break
        print(line.strip())
```

### 读取指定大小

```python
with open("test.txt", "r", encoding="utf-8") as file:
    chunk = file.read(1024)     # 读取1024个字符
    while chunk:
        print(chunk)
        chunk = file.read(1024)
```

### 复制二进制文件

```python
def copy_file(source, destination):
    with open(source, "rb") as src:
        with open(destination, "wb") as dst:
            while True:
                chunk = src.read(4096)
                if not chunk:
                    break
                dst.write(chunk)

copy_file("source.jpg", "destination.jpg")
```

## 10.6 文件定位

### seek和tell方法

```python
with open("test.txt", "r") as file:
    # 获取当前位置
    position = file.tell()
    print(f"当前位置：{position}")
    
    # 移动到指定位置
    file.seek(10)           # 从开头偏移10个字节
    file.seek(0, 0)         # 移动到开头
    file.seek(0, 2)         # 移动到末尾
    file.seek(-10, 2)       # 从末尾向前偏移10个字节
    file.seek(10, 1)        # 从当前位置向前偏移10个字节
```

## 10.7 文件和目录操作

### os模块

```python
import os

# 获取当前工作目录
print(os.getcwd())

# 改变工作目录
os.chdir("/path/to/directory")

# 列出目录内容
print(os.listdir("."))

# 创建目录
os.mkdir("newdir")          # 创建单级目录
os.makedirs("a/b/c")        # 递归创建目录

# 删除目录
os.rmdir("emptydir")        # 删除空目录
os.removedirs("a/b/c")      # 递归删除空目录

# 删除文件
os.remove("file.txt")
os.unlink("file.txt")

# 重命名
os.rename("old.txt", "new.txt")

# 检查存在性
print(os.path.exists("file.txt"))
print(os.path.isfile("file.txt"))
print(os.path.isdir("directory"))

# 获取文件信息
print(os.path.getsize("file.txt"))
print(os.path.getmtime("file.txt"))     # 修改时间
print(os.path.getctime("file.txt"))     # 创建时间
print(os.path.getatime("file.txt"))     # 访问时间
```

### os.path模块

```python

abs_path = os.path.abspath("file.txt")

print(os.path.expanduser("~/documents"))

# 展开环境变量

```

### pathlib模块（Python 3.4+）


## 10.9 文件压缩

### zipfile模块

```python
import zipfile

# 创建zip文件
with zipfile.ZipFile("archive.zip", "w") as zf:
    zf.write("file1.txt")
    zf.write("file2.txt")

# 添加文件到zip
with zipfile.ZipFile("archive.zip", "a") as zf:
    zf.write("file3.txt", "subdir/file3.txt")   # 指定压缩包内路径

# 解压zip文件
with zipfile.ZipFile("archive.zip", "r") as zf:
    zf.extractall("extracted")                  # 解压到目录
    zf.extract("file1.txt", "extracted")        # 解压单个文件

# 列出zip内容
with zipfile.ZipFile("archive.zip", "r") as zf:
    print(zf.namelist())
    for info in zf.infolist():
        print(f"{info.filename}: {info.file_size} bytes")
```

## 10.12 综合示例

### 示例1：文件管理器

```python
import os
import shutil
from pathlib import Path

class FileManager:
    """文件管理器"""
    
    def __init__(self, base_path="."):
        self.base_path = Path(base_path)
    
    def list_files(self, pattern="*"):
        """列出文件"""
        return list(self.base_path.glob(pattern))
    
    def create_directory(self, name):
        """创建目录"""
        path = self.base_path / name
        path.mkdir(parents=True, exist_ok=True)
        return path
    
    def delete(self, name):
        """删除文件或目录"""
        path = self.base_path / name
        if path.is_file():
            path.unlink()
        elif path.is_dir():
            shutil.rmtree(path)
    
    def copy(self, source, destination):
        """复制文件或目录"""
        src = self.base_path / source
        dst = self.base_path / destination
        
        if src.is_file():
            shutil.copy2(src, dst)
        elif src.is_dir():
            shutil.copytree(src, dst)
    
    def move(self, source, destination):
        """移动文件或目录"""
        src = self.base_path / source
        dst = self.base_path / destination
        shutil.move(src, dst)
    
    def get_size(self, name):
        """获取文件大小"""
        path = self.base_path / name
        if path.is_file():
            return path.stat().st_size
        elif path.is_dir():
            total = 0
            for item in path.rglob("*"):
                if item.is_file():
                    total += item.stat().st_size
            return total
    
    def search(self, pattern):
        """搜索文件"""
        results = []
        for item in self.base_path.rglob(pattern):
            results.append(item.relative_to(self.base_path))
        return results

# 使用
fm = FileManager(".")
print(fm.list_files("*.py"))
print(fm.search("*.txt"))
```

### 示例2：日志文件处理器

```python
import os
import gzip
import shutil
from datetime import datetime, timedelta
from pathlib import Path

class LogRotator:
    """日志轮转器"""
    
    def __init__(self, log_dir, max_size=10*1024*1024, max_days=7):
        self.log_dir = Path(log_dir)
        self.max_size = max_size          # 最大文件大小（字节）
        self.max_days = max_days          # 最大保留天数
        self.log_dir.mkdir(parents=True, exist_ok=True)
    
    def rotate(self, log_file):
        """轮转日志文件"""
        log_path = self.log_dir / log_file
        
        if not log_path.exists():
            return
        
        # 检查文件大小
        if log_path.stat().st_size < self.max_size:
            return
        
        # 生成备份文件名
        timestamp = datetime.now().strftime("%Y%m%d_%H%M%S")
        backup_name = f"{log_file}.{timestamp}"
        backup_path = self.log_dir / backup_name
        
        # 重命名当前日志文件
        shutil.move(log_path, backup_path)
        
        # 压缩备份文件
        with open(backup_path, "rb") as f_in:
            with gzip.open(f"{backup_path}.gz", "wb") as f_out:
                shutil.copyfileobj(f_in, f_out)
        backup_path.unlink()
        
        # 创建新的日志文件
        log_path.touch()
    
    def cleanup(self):
        """清理过期日志"""
        cutoff = datetime.now() - timedelta(days=self.max_days)
        
        for item in self.log_dir.iterdir():
            if item.is_file():
                mtime = datetime.fromtimestamp(item.stat().st_mtime)
                if mtime < cutoff:
                    item.unlink()
                    print(f"删除过期日志：{item.name}")

# 使用
rotator = LogRotator("logs")
rotator.rotate("app.log")
rotator.cleanup()
```
