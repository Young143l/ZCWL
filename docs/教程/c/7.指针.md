# 第七章 指针

## 7.1 指针概述

指针是C语言的核心特性之一，它是存储内存地址的变量。通过指针，程序可以直接访问和操作内存，实现高效的数据处理。指针广泛应用于动态内存分配、数组操作、函数参数传递和复杂数据结构的实现。

## 7.2 指针的基本概念

### 指针的定义

```c
int *p;                 /* 定义指向int类型的指针 */
double *dp;             /* 定义指向double类型的指针 */
char *cp;               /* 定义指向char类型的指针 */
void *vp;               /* 定义通用指针 */
```

### 取地址运算符与解引用运算符

```c
int a = 10;
int *p = &a;            /* &取a的地址，赋值给p */

int b = *p;             /* *解引用p，获取p指向的值，b = 10 */
*p = 20;                /* 通过p修改a的值，a变为20 */
```

### 指针的大小

指针的大小取决于系统架构，与指向的数据类型无关：

```c
printf("sizeof(int*) = %zu\n", sizeof(int*));       /* 32位系统为4，64位系统为8 */
printf("sizeof(char*) = %zu\n", sizeof(char*));     /* 同上 */
printf("sizeof(void*) = %zu\n", sizeof(void*));     /* 同上 */
```

## 7.3 指针与数组

### 数组名与指针

数组名在大多数表达式中会退化为指向首元素的指针：

```c
int arr[5] = {1, 2, 3, 4, 5};
int *p = arr;           /* 等价于 int *p = &arr[0]; */

/* 以下等价 */
printf("%d\n", arr[0]);
printf("%d\n", *arr);
printf("%d\n", *p);
printf("%d\n", p[0]);
```

### 指针运算

指针可以进行加减运算，移动的单位是指向类型的大小：

```c
int arr[5] = {10, 20, 30, 40, 50};
int *p = arr;

printf("%d\n", *p);      /* 10 */
p++;                    /* p指向arr[1] */
printf("%d\n", *p);      /* 20 */
printf("%d\n", *(p + 2)); /* 40，p + 2指向arr[3] */

/* 指针相减得到元素个数 */
int *q = &arr[4];
printf("%td\n", q - p);   /* 3，p和q之间有3个int元素 */
```

### 数组下标与指针的关系

```c
arr[i] 等价于 *(arr + i) 等价于 *(i + arr) 等价于 i[arr]
```

虽然`i[arr]`在语法上合法，但不建议使用，会降低代码可读性。

## 7.4 指针与字符串

字符串常量返回指向其首字符的指针：

```c
char *str = "Hello";    /* str指向字符串常量 */
char arr[] = "Hello";   /* arr是数组，内容可修改 */

/* 遍历字符串 */
char *p = str;
while (*p != '\0') {
    printf("%c", *p);
    p++;
}
```

## 7.5 指针作为函数参数

指针可以实现函数对调用者变量的修改：

```c
/* 交换两个整数 */
void swap(int *a, int *b)
{
    int temp = *a;
    *a = *b;
    *b = temp;
}

int main(void)
{
    int x = 3, y = 5;
    swap(&x, &y);
    printf("x = %d, y = %d\n", x, y);    /* x = 5, y = 3 */
    return 0;
}
```

### 传递数组

```c
/* 计算数组元素之和 */
int sum(int *arr, int n)
{
    int total = 0;
    for (int i = 0; i < n; i++) {
        total += arr[i];    /* 或 *(arr + i) */
    }
    return total;
}

/* 等价写法 */
int sum2(int arr[], int n)
int sum3(int arr[10], int n)    /* 10被忽略 */
```

## 7.6 指针的指针

指针的指针存储指针变量的地址：

```c
int a = 10;
int *p = &a;
int **pp = &p;

printf("%d\n", a);       /* 10 */
printf("%d\n", *p);      /* 10 */
printf("%d\n", **pp);    /* 10 */

printf("%p\n", (void*)&a);   /* a的地址 */
printf("%p\n", (void*)p);    /* a的地址 */
printf("%p\n", (void*)*pp);  /* a的地址 */
```

### 应用：修改指针本身

```c
void allocate(int **pp, int n)
{
    *pp = malloc(n * sizeof(int));
}

int main(void)
{
    int *arr;
    allocate(&arr, 10);     /* arr指向分配的内存 */
    /* 使用arr... */
    free(arr);
    return 0;
}
```

## 7.7 函数指针

函数指针存储函数的入口地址：

```c
/* 声明函数指针 */
int (*func_ptr)(int, int);

/* 定义函数 */
int add(int a, int b)
{
    return a + b;
}

/* 使用 */
func_ptr = add;             /* 或 func_ptr = &add; */
int result = func_ptr(3, 5);    /* 或 (*func_ptr)(3, 5); */
```

### 函数指针数组

```c
int add(int a, int b) { return a + b; }
int sub(int a, int b) { return a - b; }
int mul(int a, int b) { return a * b; }
int div(int a, int b) { return b != 0 ? a / b : 0; }

int (*operations[])(int, int) = {add, sub, mul, div};

int result = operations[0](10, 5);  /* 调用add，结果为15 */
```

## 7.8 动态内存分配

### malloc函数

```c
#include <stdlib.h>

/* 分配内存 */
int *arr = malloc(10 * sizeof(int));
if (arr == NULL) {
    /* 分配失败处理 */
}

/* 使用内存 */
for (int i = 0; i < 10; i++) {
    arr[i] = i;
}

/* 释放内存 */
free(arr);
arr = NULL;             /* 避免野指针 */
```

### calloc函数

```c
/* 分配并初始化为0 */
int *arr = calloc(10, sizeof(int));     /* 10个int，全部初始化为0 */
```

### realloc函数

```c
/* 重新分配内存 */
int *new_arr = realloc(arr, 20 * sizeof(int));
if (new_arr != NULL) {
    arr = new_arr;      /* 更新指针 */
}
```

## 7.9 常见指针问题

### 野指针

指向已释放内存或未初始化内存的指针：

```c
int *p;                 /* 未初始化，野指针 */
*p = 10;                /* 危险：未定义行为 */

int *q = malloc(sizeof(int));
free(q);
*q = 10;                /* 危险：使用已释放内存 */
```

### 内存泄漏

分配的内存未释放：

```c
void leak(void)
{
    int *p = malloc(100 * sizeof(int));
    /* 使用p... */
    /* 忘记free(p) */
}
```

### 空指针解引用

```c
int *p = NULL;
*p = 10;                /* 危险：空指针解引用 */
```

使用前应检查：

```c
if (p != NULL) {
    *p = 10;
}
```

## 7.10 综合示例

### 示例1：动态数组

```c
#include <stdio.h>
#include <stdlib.h>

int main(void)
{
    int n;
    printf("请输入数组大小：");
    scanf("%d", &n);
    
    /* 动态分配数组 */
    int *arr = malloc(n * sizeof(int));
    if (arr == NULL) {
        printf("内存分配失败\n");
        return 1;
    }
    
    /* 输入数据 */
    printf("请输入%d个整数：\n", n);
    for (int i = 0; i < n; i++) {
        scanf("%d", &arr[i]);
    }
    
    /* 计算平均值 */
    double sum = 0;
    for (int i = 0; i < n; i++) {
        sum += arr[i];
    }
    printf("平均值：%.2f\n", sum / n);
    
    /* 释放内存 */
    free(arr);
    
    return 0;
}
```

### 示例2：链表实现

```c
#include <stdio.h>
#include <stdlib.h>

typedef struct Node {
    int data;
    struct Node *next;
} Node;

/* 创建新节点 */
Node *create_node(int data)
{
    Node *node = malloc(sizeof(Node));
    if (node != NULL) {
        node->data = data;
        node->next = NULL;
    }
    return node;
}

/* 在链表头部插入 */
void insert_head(Node **head, int data)
{
    Node *new_node = create_node(data);
    if (new_node != NULL) {
        new_node->next = *head;
        *head = new_node;
    }
}

/* 删除节点 */
void delete_node(Node **head, int data)
{
    Node *current = *head;
    Node *prev = NULL;
    
    while (current != NULL) {
        if (current->data == data) {
            if (prev == NULL) {
                *head = current->next;
            } else {
                prev->next = current->next;
            }
            free(current);
            return;
        }
        prev = current;
        current = current->next;
    }
}

/* 打印链表 */
void print_list(Node *head)
{
    Node *current = head;
    while (current != NULL) {
        printf("%d -> ", current->data);
        current = current->next;
    }
    printf("NULL\n");
}

/* 释放链表 */
void free_list(Node *head)
{
    Node *current = head;
    while (current != NULL) {
        Node *temp = current;
        current = current->next;
        free(temp);
    }
}

int main(void)
{
    Node *head = NULL;
    
    insert_head(&head, 3);
    insert_head(&head, 2);
    insert_head(&head, 1);
    
    printf("链表：");
    print_list(head);
    
    delete_node(&head, 2);
    printf("删除2后：");
    print_list(head);
    
    free_list(head);
    
    return 0;
}
```

### 示例3：回调函数

```c
#include <stdio.h>

/* 回调函数类型 */
typedef int (*CompareFunc)(const void *, const void *);

/* 通用冒泡排序 */
void bubble_sort(void *arr, int n, size_t size, CompareFunc cmp)
{
    char *array = arr;
    char temp[size];
    
    for (int i = 0; i < n - 1; i++) {
        for (int j = 0; j < n - i - 1; j++) {
            char *elem1 = array + j * size;
            char *elem2 = array + (j + 1) * size;
            
            if (cmp(elem1, elem2) > 0) {
                /* 交换 */
                memcpy(temp, elem1, size);
                memcpy(elem1, elem2, size);
                memcpy(elem2, temp, size);
            }
        }
    }
}

/* 比较函数 */
int compare_int(const void *a, const void *b)
{
    return (*(int*)a - *(int*)b);
}

int compare_double(const void *a, const void *b)
{
    double diff = (*(double*)a - *(double*)b);
    return (diff > 0) - (diff < 0);
}

int main(void)
{
    int arr[] = {64, 34, 25, 12, 22, 11, 90};
    int n = sizeof(arr) / sizeof(arr[0]);
    
    bubble_sort(arr, n, sizeof(int), compare_int);
    
    printf("排序后：");
    for (int i = 0; i < n; i++) {
        printf("%d ", arr[i]);
    }
    printf("\n");
    
    return 0;
}
```
