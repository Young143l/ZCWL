# 第九章 文件操作

## 9.1 文件概述

文件是存储在外部存储设备上的数据集合。C语言提供了丰富的文件操作函数，声明在`stdio.h`头文件中。文件操作主要包括打开文件、读写文件和关闭文件三个基本步骤。

## 9.2 文件类型

### 文本文件

文本文件以字符形式存储数据，每个字符对应一个ASCII码。文本文件可以被人直接阅读，但存储效率较低。

### 二进制文件

二进制文件以内存中的原始形式存储数据，存储效率高，读写速度快，但不可直接阅读。

## 9.3 文件指针

C语言使用`FILE`类型的指针来操作文件：

```c
FILE *fp;                   /* 文件指针 */
```

`FILE`是标准库定义的结构体类型，包含文件的各种信息。

## 9.4 文件的打开与关闭

### 打开文件

```c
FILE *fopen(const char *filename, const char *mode);
```

打开模式：

| 模式 | 含义 |
|------|------|
| "r" | 只读，文件必须存在 |
| "w" | 只写，文件不存在则创建，存在则清空 |
| "a" | 追加，文件不存在则创建，存在则在末尾添加 |
| "r+" | 读写，文件必须存在 |
| "w+" | 读写，文件不存在则创建，存在则清空 |
| "a+" | 读写追加，文件不存在则创建 |
| "rb" | 二进制只读 |
| "wb" | 二进制只写 |
| "ab" | 二进制追加 |

```c
FILE *fp = fopen("data.txt", "r");
if (fp == NULL) {
    printf("无法打开文件\n");
    return 1;
}
```

### 关闭文件

```c
int fclose(FILE *fp);
```

```c
fclose(fp);
```

## 9.5 文本文件的读写

### 字符读写

```c
/* 写入字符 */
int fputc(int c, FILE *fp);

/* 读取字符 */
int fgetc(FILE *fp);
```

```c
/* 写入字符 */
FILE *fp = fopen("output.txt", "w");
if (fp != NULL) {
    fputc('A', fp);
    fputc('B', fp);
    fclose(fp);
}

/* 读取字符 */
FILE *fp = fopen("input.txt", "r");
if (fp != NULL) {
    int c;
    while ((c = fgetc(fp)) != EOF) {
        putchar(c);
    }
    fclose(fp);
}
```

### 字符串读写

```c
/* 写入字符串 */
int fputs(const char *s, FILE *fp);

/* 读取字符串 */
char *fgets(char *s, int n, FILE *fp);
```

```c
/* 写入字符串 */
FILE *fp = fopen("output.txt", "w");
if (fp != NULL) {
    fputs("Hello, World!\n", fp);
    fputs("This is a test.\n", fp);
    fclose(fp);
}

/* 读取字符串 */
FILE *fp = fopen("input.txt", "r");
if (fp != NULL) {
    char buffer[100];
    while (fgets(buffer, sizeof(buffer), fp) != NULL) {
        printf("%s", buffer);
    }
    fclose(fp);
}
```

### 格式化读写

```c
/* 格式化写入 */
int fprintf(FILE *fp, const char *format, ...);

/* 格式化读取 */
int fscanf(FILE *fp, const char *format, ...);
```

```c
/* 格式化写入 */
FILE *fp = fopen("data.txt", "w");
if (fp != NULL) {
    fprintf(fp, "Name: %s\n", "Alice");
    fprintf(fp, "Age: %d\n", 20);
    fprintf(fp, "Score: %.2f\n", 85.5);
    fclose(fp);
}

/* 格式化读取 */
FILE *fp = fopen("data.txt", "r");
if (fp != NULL) {
    char name[50];
    int age;
    float score;
    fscanf(fp, "Name: %s", name);
    fscanf(fp, "Age: %d", &age);
    fscanf(fp, "Score: %f", &score);
    printf("%s, %d, %.2f\n", name, age, score);
    fclose(fp);
}
```

## 9.6 二进制文件的读写

```c
/* 二进制写入 */
size_t fwrite(const void *ptr, size_t size, size_t nmemb, FILE *fp);

/* 二进制读取 */
size_t fread(void *ptr, size_t size, size_t nmemb, FILE *fp);
```

```c
/* 写入结构体数组 */
typedef struct {
    int id;
    char name[50];
    float score;
} Student;

Student students[3] = {
    {1, "Alice", 85.5},
    {2, "Bob", 90.0},
    {3, "Charlie", 78.5}
};

FILE *fp = fopen("students.dat", "wb");
if (fp != NULL) {
    fwrite(students, sizeof(Student), 3, fp);
    fclose(fp);
}

/* 读取结构体数组 */
Student read_students[3];
FILE *fp = fopen("students.dat", "rb");
if (fp != NULL) {
    fread(read_students, sizeof(Student), 3, fp);
    for (int i = 0; i < 3; i++) {
        printf("%d, %s, %.2f\n", 
               read_students[i].id, 
               read_students[i].name, 
               read_students[i].score);
    }
    fclose(fp);
}
```

## 9.7 文件定位

### 获取当前位置

```c
long ftell(FILE *fp);
```

### 移动文件指针

```c
int fseek(FILE *fp, long offset, int whence);
```

whence参数：
- `SEEK_SET`：从文件开头
- `SEEK_CUR`：从当前位置
- `SEEK_END`：从文件末尾

### 回到文件开头

```c
void rewind(FILE *fp);
```

```c
FILE *fp = fopen("data.txt", "r");
if (fp != NULL) {
    /* 移动到第10个字节 */
    fseek(fp, 10, SEEK_SET);
    
    /* 获取当前位置 */
    long pos = ftell(fp);
    printf("当前位置：%ld\n", pos);
    
    /* 从当前位置向后移动5个字节 */
    fseek(fp, 5, SEEK_CUR);
    
    /* 回到文件开头 */
    rewind(fp);
    
    fclose(fp);
}
```

## 9.8 文件状态检测

### 检测文件结束

```c
int feof(FILE *fp);
```

```c
FILE *fp = fopen("data.txt", "r");
if (fp != NULL) {
    int c;
    while ((c = fgetc(fp)) != EOF) {
        putchar(c);
    }
    
    if (feof(fp)) {
        printf("\n已到达文件末尾\n");
    }
    
    fclose(fp);
}
```

### 检测错误

```c
int ferror(FILE *fp);
void clearerr(FILE *fp);
```

```c
FILE *fp = fopen("readonly.txt", "r");
if (fp != NULL) {
    fputc('A', fp);         /* 尝试写入只读文件 */
    
    if (ferror(fp)) {
        printf("文件操作错误\n");
        clearerr(fp);       /* 清除错误标志 */
    }
    
    fclose(fp);
}
```

## 9.10 标准流

C语言预定义了三个标准流：

```c
FILE *stdin;                /* 标准输入 */
FILE *stdout;               /* 标准输出 */
FILE *stderr;               /* 标准错误 */
```

```c
/* 从标准输入读取 */

/* 写入标准输出 */

/* 写入标准错误 */
fprintf(stderr, "Error message\n");
```

## 9.11 文件操作的最佳实践

### 检查文件是否成功打开

```c
FILE *fp = fopen("data.txt", "r");
if (fp == NULL) {
    perror("无法打开文件");   /* 输出错误信息 */
    return 1;
}
```

### 确保文件关闭

```c
FILE *fp = fopen("data.txt", "r");
if (fp != NULL) {
    /* 文件操作... */
    
    if (fclose(fp) != 0) {
        perror("关闭文件失败");
    }
}
```

### 使用缓冲区提高效率

```c
/* 设置缓冲区 */
char buffer[BUFSIZ];
setvbuf(fp, buffer, _IOFBF, BUFSIZ);    /* 全缓冲 */
setvbuf(fp, NULL, _IOLBF, 0);           /* 行缓冲 */
setvbuf(fp, NULL, _IONBF, 0);           /* 无缓冲 */
```

## 9.12 综合示例

### 示例1：文件复制

```c
#include <stdio.h>

int main(int argc, char *argv[])
{
    if (argc != 3) {
        printf("用法: %s <源文件> <目标文件>\n", argv[0]);
        return 1;
    }
    
    FILE *src = fopen(argv[1], "rb");
    if (src == NULL) {
        perror("无法打开源文件");
        return 1;
    }
    
    FILE *dst = fopen(argv[2], "wb");
    if (dst == NULL) {
        perror("无法创建目标文件");
        fclose(src);
        return 1;
    }
    
    /* 复制文件 */
    char buffer[4096];
    size_t n;
    while ((n = fread(buffer, 1, sizeof(buffer), src)) > 0) {
        if (fwrite(buffer, 1, n, dst) != n) {
            perror("写入失败");
            break;
        }
    }
    
    fclose(src);
    fclose(dst);
    
    printf("文件复制完成\n");
    return 0;
}
```

### 示例2：学生信息管理系统

```c
#include <stdio.h>
#include <string.h>

#define FILENAME "students.dat"

typedef struct {
    int id;
    char name[50];
    float score;
} Student;

/* 添加学生 */
void add_student(void)
{
    FILE *fp = fopen(FILENAME, "ab");
    if (fp == NULL) {
        perror("无法打开文件");
        return;
    }
    
    Student s;
    printf("请输入学生ID：");
    scanf("%d", &s.id);
    printf("请输入姓名：");
    scanf("%s", s.name);
    printf("请输入成绩：");
    scanf("%f", &s.score);
    
    fwrite(&s, sizeof(Student), 1, fp);
    fclose(fp);
    
    printf("学生信息已保存\n");
}

/* 显示所有学生 */
void display_all(void)
{
    FILE *fp = fopen(FILENAME, "rb");
    if (fp == NULL) {
        printf("暂无学生信息\n");
        return;
    }
    
    Student s;
    printf("\n学生信息列表：\n");
    printf("%-10s %-20s %-10s\n", "ID", "姓名", "成绩");
    printf("----------------------------------------\n");
    
    while (fread(&s, sizeof(Student), 1, fp) == 1) {
        printf("%-10d %-20s %-10.2f\n", s.id, s.name, s.score);
    }
    
    fclose(fp);
}

/* 查找学生 */
void search_student(void)
{
    int id;
    printf("请输入要查找的学生ID：");
    scanf("%d", &id);
    
    FILE *fp = fopen(FILENAME, "rb");
    if (fp == NULL) {
        printf("文件不存在\n");
        return;
    }
    
    Student s;
    int found = 0;
    while (fread(&s, sizeof(Student), 1, fp) == 1) {
        if (s.id == id) {
            printf("找到学生：\n");
            printf("ID: %d\n", s.id);
            printf("姓名: %s\n", s.name);
            printf("成绩: %.2f\n", s.score);
            found = 1;
            break;
        }
    }
    
    if (!found) {
        printf("未找到该学生\n");
    }
    
    fclose(fp);
}

int main(void)
{
    int choice;
    
    do {
        printf("\n学生信息管理系统\n");
        printf("1. 添加学生\n");
        printf("2. 显示所有学生\n");
        printf("3. 查找学生\n");
        printf("0. 退出\n");
        printf("请选择：");
        scanf("%d", &choice);
        
        switch (choice) {
            case 1:
                add_student();
                break;
            case 2:
                display_all();
                break;
            case 3:
                search_student();
                break;
            case 0:
                printf("再见\n");
                break;
            default:
                printf("无效选择\n");
        }
    } while (choice != 0);
    
    return 0;
}
```

### 示例3：日志文件记录

```c
#include <stdio.h>
#include <time.h>

#define LOG_FILE "app.log"

/* 写入日志 */
void write_log(const char *level, const char *message)
{
    FILE *fp = fopen(LOG_FILE, "a");
    if (fp == NULL) {
        return;
    }
    
    /* 获取当前时间 */
    time_t now = time(NULL);
    struct tm *tm_info = localtime(&now);
    char time_str[26];
    strftime(time_str, sizeof(time_str), "%Y-%m-%d %H:%M:%S", tm_info);
    
    /* 写入日志 */
    fprintf(fp, "[%s] [%s] %s\n", time_str, level, message);
    
    fclose(fp);
}

/* 读取日志 */
void read_log(void)
{
    FILE *fp = fopen(LOG_FILE, "r");
    if (fp == NULL) {
        printf("日志文件不存在\n");
        return;
    }
    
    char line[256];
    while (fgets(line, sizeof(line), fp) != NULL) {
        printf("%s", line);
    }
    
    fclose(fp);
}

int main(void)
{
    write_log("INFO", "应用程序启动");
    write_log("DEBUG", "初始化配置");
    write_log("INFO", "处理数据");
    write_log("WARNING", "检测到异常");
    write_log("ERROR", "操作失败");
    write_log("INFO", "应用程序关闭");
    
    printf("\n日志内容：\n");
    read_log();
    
    return 0;
}
```
