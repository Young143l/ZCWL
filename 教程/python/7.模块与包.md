# 第七章 模块与包

## 7.1 模块概述

模块是包含Python代码的文件，可以定义函数、类和变量。模块是组织Python代码的基本方式，有助于代码的重用和维护。

使用模块的好处：
- 代码重用：将常用功能封装在模块中，供多个程序使用
- 命名空间：避免变量名冲突
- 代码组织：将大型程序分解为可管理的部分

## 7.2 导入模块

### import语句

```python
# 导入整个模块
import math
print(math.pi)
print(math.sqrt(16))

# 导入模块中的特定内容
from math import pi, sqrt
print(pi)
print(sqrt(16))

# 使用别名
import math as m
print(m.pi)

from math import pi as PI
print(PI)

# 导入所有内容（不推荐）
from math import *
print(pi)
print(sqrt(16))
```

### 导入搜索路径

1. 当前目录
2. PYTHONPATH环境变量中的目录
3. Python标准库目录
4. .pth文件中的目录

```python

```

## 7.3 创建模块

创建模块就是创建一个.py文件：

```python
# mymath.py
"""数学工具模块"""

PI = 3.14159

def add(a, b):
    """返回两个数的和"""
    return a + b

def subtract(a, b):
    """返回两个数的差"""
    return a - b

def multiply(a, b):
    """返回两个数的积"""
    return a * b

def divide(a, b):
    """返回两个数的商"""
    if b == 0:
        raise ValueError("除数不能为零")
    return a / b

class Calculator:
    """简单计算器类"""
    
    def __init__(self):
        self.history = []
    
    def calculate(self, a, b, operation):
        """执行计算"""
        if operation == "+":
            result = add(a, b)
        elif operation == "-":
            result = subtract(a, b)
        elif operation == "*":
            result = multiply(a, b)
        elif operation == "/":
            result = divide(a, b)
        else:
            raise ValueError("不支持的运算")
        
        self.history.append(f"{a} {operation} {b} = {result}")
        return result
```

使用自定义模块：

```python
import mymath

print(mymath.PI)
print(mymath.add(3, 5))

calc = mymath.Calculator()
print(calc.calculate(10, 5, "+"))
```

## 7.4 模块执行

### __name__属性

```python
# mymodule.py

```

当模块被直接运行时，`__name__`的值为`"__main__"`；

## 7.5 包

包是包含多个模块的目录，必须包含`__init__.py`文件（Python 3.3+可选）。

### 包的结构

```
mypackage/
    __init__.py
    module1.py
    module2.py
    subpackage/
        __init__.py
        module3.py
```

### 创建包

```python
# mypackage/__init__.py
"""我的包"""

__version__ = "1.0.0"

from .module1 import func1
from .module2 import func2

__all__ = ["func1", "func2"]
```

```python
# mypackage/module1.py
def func1():
    return "Function 1"
```

```python
# mypackage/module2.py
def func2():
    return "Function 2"
```

### 使用包

```python
# 导入包
import mypackage
print(mypackage.func1())

# 导入子模块
from mypackage import module1
print(module1.func1())

# 导入特定函数
from mypackage.module1 import func1
print(func1())

# 导入子包
from mypackage.subpackage import module3
```

### 相对导入

```python
# mypackage/subpackage/module3.py
from ..module1 import func1      # 从父包导入
from . import module4            # 从同级目录导入
```

### re模块

正则表达式：

```python
import re

# 匹配
text = "Hello, World!"
match = re.search(r"World", text)
if match:
    print("Found:", match.group())

# 查找所有
numbers = re.findall(r"\d+", "abc123def456")
print(numbers)                  # ['123', '456']

# 替换
result = re.sub(r"World", "Python", text)
print(result)                   # Hello, Python!

# 分割
parts = re.split(r"[,!]", text)
print(parts)                    # ['Hello', ' World', '']

# 编译正则表达式
pattern = re.compile(r"\d+")
matches = pattern.findall("abc123def456")

# 分组
match = re.search(r"(\d{4})-(\d{2})-(\d{2})", "2024-01-15")
if match:
    print(match.group(0))       # 2024-01-15
    print(match.group(1))       # 2024
    print(match.group(2))       # 01
    print(match.group(3))       # 15
```

### random模块

随机数生成：

```python
import random

# 随机整数
print(random.randint(1, 10))        # 1到10的随机整数
print(random.randrange(1, 10, 2))   # 1到9的奇数

# 随机浮点数
print(random.random())              # 0到1的随机浮点数
print(random.uniform(1, 10))        # 1到10的随机浮点数

# 随机选择
choices = ["apple", "banana", "cherry"]
print(random.choice(choices))
print(random.choices(choices, k=2))     # 可重复选择
print(random.sample(choices, k=2))      # 不重复选择

# 打乱顺序
random.shuffle(choices)
print(choices)

# 设置随机种子（可重现）
random.seed(42)
```

## 7.8 综合示例

### 示例1：配置文件管理模块

```python
# config.py
"""配置文件管理模块"""

import json
import os

class Config:
    """配置文件管理类"""
    
    def __init__(self, filename="config.json"):
        self.filename = filename
        self.data = {}
        self.load()
    
    def load(self):
        """加载配置文件"""
        if os.path.exists(self.filename):
            with open(self.filename, "r", encoding="utf-8") as f:
                self.data = json.load(f)
    
    def save(self):
        """保存配置文件"""
        with open(self.filename, "w", encoding="utf-8") as f:
            json.dump(self.data, f, indent=4, ensure_ascii=False)
    
    def get(self, key, default=None):
        """获取配置项"""
        return self.data.get(key, default)
    
    def set(self, key, value):
        """设置配置项"""
        self.data[key] = value
        self.save()
    
    def delete(self, key):
        """删除配置项"""
        if key in self.data:
            del self.data[key]
            self.save()

# 使用示例
if __name__ == "__main__":
    config = Config()
    
    config.set("app_name", "MyApp")
    config.set("version", "1.0.0")
    config.set("debug", True)
    
    print(config.get("app_name"))
    print(config.get("timeout", 30))
```

### 示例2：日志模块

```python
# logger.py
"""简单日志模块"""

import datetime
import os

class Logger:
    """简单日志类"""
    
    LEVELS = {
        "DEBUG": 10,
        "INFO": 20,
        "WARNING": 30,
        "ERROR": 40,
        "CRITICAL": 50
    }
    
    def __init__(self, filename=None, level="INFO"):
        self.filename = filename
        self.level = self.LEVELS.get(level, 20)
    
    def _log(self, level, message):
        """记录日志"""
        if self.LEVELS.get(level, 0) < self.level:
            return
        
        timestamp = datetime.datetime.now().strftime("%Y-%m-%d %H:%M:%S")
        log_entry = f"[{timestamp}] [{level}] {message}"
        
        print(log_entry)
        
        if self.filename:
            with open(self.filename, "a", encoding="utf-8") as f:
                f.write(log_entry + "\n")
    
    def debug(self, message):
        self._log("DEBUG", message)
    
    def info(self, message):
        self._log("INFO", message)
    
    def warning(self, message):
        self._log("WARNING", message)
    
    def error(self, message):
        self._log("ERROR", message)
    
    def critical(self, message):
        self._log("CRITICAL", message)

# 使用示例
if __name__ == "__main__":
    logger = Logger("app.log", "DEBUG")
    
    logger.debug("调试信息")
    logger.info("应用程序启动")
    logger.warning("警告信息")
    logger.error("错误信息")
    logger.critical("严重错误")
```
