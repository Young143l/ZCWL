# 第八章 结构体与联合体

## 8.1 结构体概述

结构体是用户自定义的数据类型，可以将不同类型的数据组合在一起。结构体在组织复杂数据、实现面向对象编程概念和构建数据结构中发挥重要作用。

## 8.2 结构体的定义与声明

### 基本定义

```c
struct Student {
    char name[50];
    int age;
    float score;
};

/* 定义结构体变量 */
struct Student stu1;
struct Student stu2 = {"Alice", 20, 85.5};
```

### 使用typedef

```c
typedef struct {
    char name[50];
    int age;
    float score;
} Student;

/* 直接使用别名 */
Student stu1;
Student stu2 = {"Alice", 20, 85.5};
```

### 定义时同时声明变量

```c
struct Point {
    int x;
    int y;
} p1, p2;                   /* 同时声明两个变量 */

struct {
    int x;
    int y;
} p3;                       /* 匿名结构体 */
```

## 8.3 结构体成员的访问

### 点运算符

```c
Student stu;
strcpy(stu.name, "Bob");
stu.age = 21;
stu.score = 90.0;

printf("姓名：%s\n", stu.name);
printf("年龄：%d\n", stu.age);
printf("成绩：%.2f\n", stu.score);
```

### 箭头运算符

```c
Student stu;
Student *p = &stu;

/* 以下等价 */
strcpy(p->name, "Bob");
strcpy((*p).name, "Bob");

p->age = 21;
(*p).age = 21;
```

## 8.4 结构体的初始化

```c
/* 顺序初始化 */
Student stu1 = {"Alice", 20, 85.5};

/* 指定成员初始化（C99） */
Student stu2 = {.name = "Bob", .age = 21};
Student stu3 = {.score = 90.0, .name = "Charlie"};

/* 部分初始化，其余为0 */
Student stu4 = {"David"};
```

## 8.5 结构体数组

```c
Student class[30];          /* 30个学生的数组 */

/* 初始化 */
Student class2[3] = {
    {"Alice", 20, 85.5},
    {"Bob", 21, 90.0},
```

## 8.6 结构体指针

```c
Student stu = {"Alice", 20, 85.5};
Student *p = &stu;

/* 访问成员 */
printf("%s\n", p->name);
printf("%d\n", p->age);

/* 通过指针修改 */
p->age = 21;
```

### 动态分配结构体

```c
Student *p = malloc(sizeof(Student));
if (p != NULL) {
    strcpy(p->name, "Alice");
    p->age = 20;
    p->score = 85.5;
    
    /* 使用... */
    
    free(p);
}
```

## 8.7 结构体作为函数参数

### 传值调用

```c
void print_student(Student s)
{
    printf("%s, %d, %.2f\n", s.name, s.age, s.score);
}

void update_score(Student s, float new_score)
{
    s.score = new_score;    /* 不影响原结构体 */
}
```

### 传址调用

```c
void update_score(Student *s, float new_score)
{
    s->score = new_score;   /* 修改原结构体 */
}

/* 使用 */
Student stu = {"Alice", 20, 85.5};
update_score(&stu, 90.0);
```

## 8.9 结构体与指针的高级应用

### 自引用结构体

### 结构体中的柔性数组

## 8.10 联合体

联合体所有成员共享同一块内存，同一时间只能存储一个成员的值：

```c
union Data {
    int i;
    float f;
    char str[20];
};

union Data d;
d.i = 10;                   /* 存储整数 */
d.f = 3.14;                 /* 现在存储浮点数，i的值被破坏 */
```

### 联合体的大小

联合体的大小等于最大成员的大小：

```c
printf("sizeof(union Data) = %zu\n", sizeof(union Data));  /* 20 */
```

### 联合体的应用

```c
/* 存储不同类型的数据 */
typedef struct {
    enum {INT, FLOAT, STRING} type;
    union {
        int i;
        float f;
        char s[20];
    } value;
} Variant;

void print_variant(Variant v)
{
    switch (v.type) {
        case INT:
            printf("%d\n", v.value.i);
            break;
        case FLOAT:
            printf("%f\n", v.value.f);
            break;
        case STRING:
            printf("%s\n", v.value.s);
            break;
    }
}
```

## 8.11 位域

位域允许在结构体中定义按位存储的成员，节省内存：

```c
struct Flags {
    unsigned int flag1 : 1;     /* 1位 */
    unsigned int flag2 : 1;     /* 1位 */
    unsigned int flag3 : 1;     /* 1位 */
    unsigned int value : 4;     /* 4位 */
    unsigned int : 2;           /* 2位填充 */
    unsigned int data : 8;      /* 8位 */
};

struct Flags f = {1, 0, 1, 15, 255};
printf("sizeof(struct Flags) = %zu\n", sizeof(struct Flags));  /* 通常4字节 */
```

## 8.12 结构体的内存对齐

为了提高访问效率，编译器会对结构体进行内存对齐：

```c
struct Example {
    char c;         /* 1字节 */
    int i;          /* 4字节，通常对齐到4字节边界 */
    char d;         /* 1字节 */
};

/* 实际大小可能为12字节，而非6字节 */
```

可以使用`#pragma pack`修改对齐方式：

```c
#pragma pack(1)     /* 按1字节对齐 */
struct Packed {
    char c;
    int i;
    char d;
};
#pragma pack()      /* 恢复默认对齐 */

/* 现在大小为6字节 */
```

## 8.13 综合示例

### 示例1：学生管理系统

```c
#include <stdio.h>
#include <string.h>

#define MAX_STUDENTS 100

typedef struct {
    int id;
    char name[50];
    int age;
    float scores[3];        /* 三门课程成绩 */
} Student;

/* 计算平均分 */
float average_score(const Student *s)
{
    float sum = 0;
    for (int i = 0; i < 3; i++) {
        sum += s->scores[i];
    }
    return sum / 3;
}

/* 打印学生信息 */
void print_student(const Student *s)
{
    printf("ID: %d, 姓名: %s, 年龄: %d\n", s->id, s->name, s->age);
    printf("成绩: ");
    for (int i = 0; i < 3; i++) {
        printf("%.2f ", s->scores[i]);
    }
    printf("\n平均分: %.2f\n\n", average_score(s));
}

int main(void)
{
    Student students[MAX_STUDENTS];
    int count = 0;
    
    /* 添加学生 */
    students[count++] = (Student){1, "Alice", 20, {85.5, 90.0, 88.5}};
    students[count++] = (Student){2, "Bob", 21, {78.0, 82.5, 80.0}};
    students[count++] = (Student){3, "Charlie", 19, {92.0, 95.5, 91.0}};
    
    /* 打印所有学生 */
    printf("学生信息：\n");
    for (int i = 0; i < count; i++) {
        print_student(&students[i]);
    }
    
    return 0;
}
```

### 示例2：二叉树实现

```c
#include <stdio.h>
#include <stdlib.h>

typedef struct TreeNode {
    int data;
    struct TreeNode *left;
    struct TreeNode *right;
} TreeNode;

/* 创建节点 */
TreeNode *create_node(int data)
{
    TreeNode *node = malloc(sizeof(TreeNode));
    if (node != NULL) {
        node->data = data;
        node->left = NULL;
        node->right = NULL;
    }
    return node;
}

/* 插入节点 */
TreeNode *insert(TreeNode *root, int data)
{
    if (root == NULL) {
        return create_node(data);
    }
    
    if (data < root->data) {
        root->left = insert(root->left, data);
    } else if (data > root->data) {
        root->right = insert(root->right, data);
    }
    
    return root;
}

/* 中序遍历 */
void inorder(TreeNode *root)
{
    if (root != NULL) {
        inorder(root->left);
        printf("%d ", root->data);
        inorder(root->right);
    }
}

/* 释放树 */
void free_tree(TreeNode *root)
{
    if (root != NULL) {
        free_tree(root->left);
        free_tree(root->right);
        free(root);
    }
}

int main(void)
{
    TreeNode *root = NULL;
    int values[] = {50, 30, 70, 20, 40, 60, 80};
    int n = sizeof(values) / sizeof(values[0]);
    
    for (int i = 0; i < n; i++) {
        root = insert(root, values[i]);
    }
    
    printf("中序遍历：");
    inorder(root);
    printf("\n");
    
    free_tree(root);
    
    return 0;
}
```

### 示例3：使用联合体处理网络协议

```c
#include <stdio.h>
#include <stdint.h>

/* IP地址联合体 */
union IPAddress {
    uint32_t addr;
    uint8_t bytes[4];
};

/* 打印IP地址 */
void print_ip(union IPAddress ip)
{
    printf("%d.%d.%d.%d\n", 
           ip.bytes[0], ip.bytes[1], ip.bytes[2], ip.bytes[3]);
    printf("32位值: 0x%08X\n", ip.addr);
}

int main(void)
{
    union IPAddress ip;
    ip.addr = 0xC0A80101;       /* 192.168.1.1 */
    
    printf("IP地址：");
    print_ip(ip);
    
    /* 修改单个字节 */
    ip.bytes[3] = 100;
    printf("修改后：");
    print_ip(ip);
    
    return 0;
}
```
