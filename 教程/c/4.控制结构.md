# 第四章 控制结构

## 4.1 程序控制概述

程序的控制结构决定了代码的执行顺序。C语言提供了三种基本控制结构：顺序结构、选择结构和循环结构。顺序结构按照代码的先后顺序依次执行；选择结构根据条件判断执行不同的代码分支；循环结构重复执行某段代码直到满足退出条件。

## 4.2 选择结构

### if语句

if语句是最基本的选择结构：

```c
if (条件表达式) {
    /* 条件为真时执行的语句 */
}
```

```c
int score = 85;
if (score >= 60) {
    printf("及格\n");
}
```

### if-else语句

if-else语句提供两个分支选择：

```c
if (条件表达式) {
    /* 条件为真时执行 */
} else {
    /* 条件为假时执行 */
}
```

```c
int score = 55;
if (score >= 60) {
    printf("及格\n");
} else {
    printf("不及格\n");
}
```

### if-else if-else语句

多个条件的判断：

```c
if (条件1) {
    /* 条件1为真 */
} else if (条件2) {
    /* 条件1为假，条件2为真 */
} else if (条件3) {
    /* 条件1、2为假，条件3为真 */
} else {
    /* 所有条件都为假 */
}
```

```c
int score = 85;
if (score >= 90) {
    printf("优秀\n");
} else if (score >= 80) {
    printf("良好\n");
} else if (score >= 70) {
    printf("中等\n");
} else if (score >= 60) {
    printf("及格\n");
} else {
    printf("不及格\n");
}
```

### 嵌套if语句

if语句可以嵌套使用：

```c
int age = 25;
int income = 50000;

if (age >= 18) {
    if (income >= 30000) {
        printf("符合贷款条件\n");
    } else {
        printf("收入不足\n");
    }
} else {
    printf("年龄不符合要求\n");
}
```

### switch语句

switch语句用于多分支选择，适用于离散值的判断：

```c
switch (表达式) {
    case 常量1:
        /* 语句 */
        break;
    case 常量2:
        /* 语句 */
        break;
    default:
        /* 默认语句 */
}
```

```c
int day = 3;
switch (day) {
    case 1:
        printf("星期一\n");
        break;
    case 2:
        printf("星期二\n");
        break;
    case 3:
        printf("星期三\n");
        break;
    case 4:
        printf("星期四\n");
        break;
    case 5:
        printf("星期五\n");
        break;
    case 6:
    case 7:
        printf("周末\n");
        break;
    default:
        printf("无效的日期\n");
}
```

注意：
- switch表达式的值必须是整型或字符型
- case标签必须是常量表达式
- 每个case通常以break结束，否则会继续执行下一个case（穿透）
- default分支是可选的

## 4.3 循环结构

### while循环

while循环先判断条件，条件为真时执行循环体：

```c
while (条件表达式) {
    /* 循环体 */
}
```

```c
/* 计算1到100的和 */
int i = 1;
int sum = 0;
while (i <= 100) {
    sum += i;
    i++;
}
printf("1到100的和：%d\n", sum);
```

### do-while循环

do-while循环先执行循环体，再判断条件，至少执行一次：

```c
do {
    /* 循环体 */
} while (条件表达式);
```

```c
/* 用户输入验证 */
int num;
do {
    printf("请输入一个正数：");
    scanf("%d", &num);
} while (num <= 0);
printf("你输入了：%d\n", num);
```

### for循环

for循环将初始化、条件判断和迭代操作集中在一行：

```c
for (初始化; 条件; 迭代) {
    /* 循环体 */
}
```

```c
/* 计算1到100的和 */
int sum = 0;
for (int i = 1; i <= 100; i++) {
    sum += i;
}
printf("1到100的和：%d\n", sum);
```

for循环的三个部分都可以省略：

```c
/* 无限循环 */
for (;;) {
    /* 循环体 */
}

/* 省略初始化 */
int i = 0;
for (; i < 10; i++) {
    /* 循环体 */
}
```

### 循环比较

三种循环的选择建议：

- **for循环**：循环次数已知，或需要计数器控制
- **while循环**：循环次数不确定，先判断后执行
- **do-while循环**：至少需要执行一次循环体

## 4.4 跳转语句

### break语句

break用于立即退出当前循环或switch语句：

```c
/* 查找数组中第一个负数 */
int arr[] = {3, 7, -2, 5, 8};
int found = -1;
for (int i = 0; i < 5; i++) {
    if (arr[i] < 0) {
        found = i;
        break;          /* 找到后立即退出循环 */
    }
}
```

### continue语句

continue跳过当前迭代，继续下一次循环：

```c
/* 打印1到10中的奇数 */
for (int i = 1; i <= 10; i++) {
    if (i % 2 == 0) {
        continue;       /* 偶数跳过 */
    }
    printf("%d ", i);
}
```

### goto语句

goto语句无条件跳转到指定标签：

```c
for (int i = 0; i < 10; i++) {
    for (int j = 0; j < 10; j++) {
        if (i * j > 50) {
            goto end;   /* 跳出多层循环 */
        }
    }
}
end:
printf("循环结束\n");
```

注意：goto语句容易导致代码难以理解和维护，应谨慎使用。

## 4.5 嵌套循环

循环可以嵌套使用，常用于处理多维数据结构：

```c
/* 打印九九乘法表 */
for (int i = 1; i <= 9; i++) {
    for (int j = 1; j <= i; j++) {
        printf("%d*%d=%-2d ", j, i, i * j);
    }
    printf("\n");
}
```

输出：
```
1*1=1  
1*2=2  2*2=4  
1*3=3  2*3=6  3*3=9  
1*4=4  2*4=8  3*4=12 4*4=16 
...
```

## 4.6 综合示例

### 示例1：判断素数

```c
#include <stdio.h>
#include <stdbool.h>
#include <math.h>

bool is_prime(int n)
{
    if (n <= 1) {
        return false;
    }
    if (n == 2) {
        return true;
    }
    if (n % 2 == 0) {
        return false;
    }
    for (int i = 3; i <= sqrt(n); i += 2) {
        if (n % i == 0) {
            return false;
        }
    }
    return true;
}

int main(void)
{
    int num;
    printf("请输入一个正整数：");
    scanf("%d", &num);
    
    if (is_prime(num)) {
        printf("%d是素数\n", num);
    } else {
        printf("%d不是素数\n", num);
    }
    
    return 0;
}
```

### 示例2：计算阶乘

```c
#include <stdio.h>

long long factorial(int n)
{
    if (n < 0) {
        return -1;          /* 负数没有阶乘 */
    }
    long long result = 1;
    for (int i = 2; i <= n; i++) {
        result *= i;
    }
    return result;
}

int main(void)
{
    int num;
    printf("请输入一个非负整数：");
    scanf("%d", &num);
    
    long long fact = factorial(num);
    if (fact < 0) {
        printf("输入无效\n");
    } else {
        printf("%d! = %lld\n", num, fact);
    }
    
    return 0;
}
```

### 示例3：斐波那契数列

```c
#include <stdio.h>

int main(void)
{
    int n;
    printf("请输入要生成的斐波那契数列项数：");
    scanf("%d", &n);
    
    if (n <= 0) {
        printf("请输入正整数\n");
        return 1;
    }
    
    long long first = 0, second = 1;
    
    printf("斐波那契数列前%d项：\n", n);
    
    for (int i = 0; i < n; i++) {
        if (i == 0) {
            printf("%lld ", first);
        } else if (i == 1) {
            printf("%lld ", second);
        } else {
            long long next = first + second;
            printf("%lld ", next);
            first = second;
            second = next;
        }
    }
    printf("\n");
    
    return 0;
}
```

### 示例4：猜数字游戏

```c
#include <stdio.h>
#include <stdlib.h>
#include <time.h>

int main(void)
{
    srand(time(NULL));
    int target = rand() % 100 + 1;      /* 1到100的随机数 */
    int guess;
    int attempts = 0;
    
    printf("猜数字游戏（1-100）\n");
    
    do {
        printf("请输入你的猜测：");
        scanf("%d", &guess);
        attempts++;
        
        if (guess > target) {
            printf("太大了！\n");
        } else if (guess < target) {
            printf("太小了！\n");
        } else {
            printf("恭喜你猜对了！用了%d次\n", attempts);
        }
    } while (guess != target);
    
    return 0;
}
```
